## Cold start is three problems, not one

Every model since ch. 3 learns a vector per user and per item from interactions; take the
interactions away and the vector does not exist. The failure has three shapes. **New item**: a
title launched this morning has no column, so collaborative filtering cannot rank it at all. **New
user**: someone who signed up a minute ago has no row, only a few attributes (country, device, the
ad they came from). **New system**: both at once — a new market or product line, where only side
information exists.

The fix is always the same idea: describe users and items by **features** that exist before any
interaction does, and learn how features map to taste. The world below gives items a text
description and users a demographic segment, so all three cases can be simulated and measured.

```python
import numpy as np
import pandas as pd
from scipy import sparse

rng = np.random.default_rng(0)
n_users, n_items, d, K, n_segments = 2_000, 800, 8, 8, 6

# items: a topic mixture (what the description says) plus a residual nobody wrote down
genre = rng.integers(0, K, n_items)
theta = rng.dirichlet(np.full(K, 0.3), n_items)
theta[np.arange(n_items), genre] += 1.0
theta /= theta.sum(axis=1, keepdims=True)
item_vec = 1.6 * theta @ rng.normal(0, 1.0, (K, d)) + rng.normal(0, 0.55, (n_items, d))
exposure = rng.lognormal(-1.0, 1.0, n_items)

topic_words = [t.split() for t in (
    "noir detective crime heist alibi clue gritty murder rain suspect",
    "space alien starship galaxy robot future planet orbit android cyber",
    "romance love wedding heart kiss letters summer longing couple date",
    "horror ghost haunted blood scream curse night demon fog cellar",
    "comedy quirky awkward prank roommate laugh office satire banter chaos",
    "war soldier battle empire sword siege history general trench medal",
    "family kids dog holiday school magic puppet cartoon friendship home",
    "documentary nature ocean climate science interview archive true wildlife food")]
generic = "story film life world new great best time people journey classic award dark bright epic".split()

def describe(t):
    words = [topic_words[k][rng.integers(0, 10)] for k in rng.choice(K, size=12, p=theta[t])]
    words += list(rng.choice(generic, 6))
    rng.shuffle(words)
    return " ".join(words)

text = [describe(t) for t in range(n_items)]

# users: a demographic segment shifts taste, the rest is individual
segment = rng.integers(0, n_segments, n_users)
user_vec = rng.normal(0, 0.6, (n_segments, d))[segment] + rng.normal(0, 1.0, (n_users, d))
activity = rng.lognormal(-0.3, 0.9, n_users)

logit = user_vec @ item_vec.T / 2.0 - 4.2 + np.log(activity)[:, None] + np.log(exposure)[None, :]
r, c = np.nonzero(rng.random((n_users, n_items)) < 1 / (1 + np.exp(-logit)))
inter = pd.DataFrame({"user": r, "item": c})
print(f"{n_users:,} users x {n_items} items | {len(inter):,} interactions "
      f"({len(inter) / (n_users * n_items):.2%} dense)")
for t in (0, 1):
    print(f"item {t} (main topic {genre[t]}): {text[t]}")
```

```output
2,000 users x 800 items | 52,827 interactions (3.30% dense)
item 0 (main topic 6): home food family school dark prank journey blood epic ocean dark dog roommate great home home bright friendship
item 1 (main topic 5): people suspect journey general epic sword battle trench war general new battle detective gritty trench story film magic
```

An item's true latent vector is **part content, part residual**: the topic mixture explains a share
of it, and the rest — quality, chemistry, whatever the description does not say — is recoverable
only from interactions. That is the realistic case, and it is why content-based and collaborative
models are complements rather than rivals.

## The evaluation protocol has to create the cold start

A random holdout of interactions does **not** test cold start: almost every held-out interaction
involves an item and a user that also appear in training. You have to hold out **entities** —
whole items and whole users, with every one of their interactions.

```python
split_rng = np.random.default_rng(1)
cold_items = np.sort(split_rng.choice(n_items, 160, replace=False))
cold_users = np.sort(split_rng.choice(n_users, 400, replace=False))
is_cold_item = np.isin(np.arange(n_items), cold_items)
is_cold_user = np.isin(np.arange(n_users), cold_users)
warm_items, warm_users = np.flatnonzero(~is_cold_item), np.flatnonzero(~is_cold_user)

ci, cu = is_cold_item[inter["item"]], is_cold_user[inter["user"]]
warm = inter[~ci & ~cu]
hold = split_rng.random(len(warm)) < 0.2
train, test_warm = warm[~hold], warm[hold]
X = sparse.csr_matrix((np.ones(len(train)), (train["user"], train["item"])), shape=(n_users, n_items))
seen = {u: set(g) for u, g in train.groupby("user")["item"]}
pop, n_u = np.asarray(X.sum(axis=0)).ravel(), np.asarray(X.sum(axis=1)).ravel()

def make_truth(df):
    return {u: set(g) for u, g in df.groupby("user")["item"]}

TASKS = {"warm": (make_truth(test_warm), warm_items),          # task -> (truth, candidates)
         "cold-item": (make_truth(inter[ci & ~cu]), cold_items),
         "cold-user": (make_truth(inter[~ci & cu]), warm_items),
         "cold-both": (make_truth(inter[ci & cu]), cold_items)}

def recall_at_k(score_fn, task, k=10, se=False, users=None):
    truth, cand = TASKS[task]
    rec = []
    for u in (truth if users is None else users):
        s = np.asarray(score_fn(u), dtype=float)[cand].copy()
        s[np.isin(cand, list(seen.get(u, ())))] = -np.inf     # never re-recommend history
        top = cand[np.argpartition(-s, k)[:k]]
        rec.append(len(truth[u] & set(top)) / min(k, len(truth[u])))
    m = float(np.mean(rec))
    return (m, float(np.std(rec) / np.sqrt(len(rec)))) if se else m

print(f"train {len(train):,} interactions")
for t, (truth, cand) in TASKS.items():
    print(f"  {t:>9}: {len(truth):>5} users, {sum(map(len, truth.values())):>5} held-out "
          f"interactions, ranked among {len(cand)} items")
r_hold = np.random.default_rng(2).random(len(inter)) < 0.2
print(f"\nrandom 20% interaction split: "
      f"{np.mean(~inter['item'][r_hold].isin(set(inter['item'][~r_hold]))):.2%} "
      f"of test interactions involve an item with no training data")
```

```output
train 26,963 interactions
       warm:  1331 users,  6708 held-out interactions, ranked among 640 items
  cold-item:  1399 users,  8525 held-out interactions, ranked among 160 items
  cold-user:   395 users,  8534 held-out interactions, ranked among 640 items
  cold-both:   352 users,  2097 held-out interactions, ranked among 160 items

random 20% interaction split: 0.00% of test interactions involve an item with no training data
```

**Warm** is the usual held-out-interaction test. **Cold-item** asks warm users to rank 160 items no
model has trained on; **cold-user** serves 400 users with zero history from the warm catalogue;
**cold-both** is the new-system case. The last line is why this matters: a random split leaves no
cold items in the test set at all, so it measures nothing about this chapter's problem.

Cold items are ranked **among cold items** — the "new arrivals" shelf — so their scores never
compete with warm items' scores. Mixing the two pools in one list is a separate calibration problem,
usually handled by reserving slots (the "mixed" hybrid below) or by exploration (ch. 11).

```python
seg_pop = np.zeros((n_segments, n_items))
np.add.at(seg_pop, (segment[train["user"].values], train["item"].values), 1.0)
tie = np.random.default_rng(5).random(n_items) * 1e-6          # breaks ties among zero counts
baselines = {"random": lambda u: np.random.default_rng(u).random(n_items),
             "popularity": lambda u: pop + tie,
             "segment popularity": lambda u: seg_pop[segment[u]] + tie}
print(f"{'recall@10':>20} " + " ".join(f"{t:>10}" for t in TASKS))
for name, fn in baselines.items():
    print(f"{name:>20} " + " ".join(f"{recall_at_k(fn, t):>10.4f}" for t in TASKS))
```

```output

              random     0.0178     0.0677     0.0446     0.0620
          popularity     0.1463     0.0716     0.2628     0.0719
  segment popularity     0.1795     0.0716     0.2897     0.0719
```

Popularity is strong on warm and cold-user tasks and **no better than random on new items**: it has
no counts for them. Segment popularity is the cheapest personalisation there is, and it already
beats global popularity for new users. Several fancier methods below fail to beat it.

## Content-based recommendation

A content-based recommender never looks at other users. It describes each item by a feature
vector, builds each user's profile from the items they consumed, and ranks by similarity.

### TF-IDF and cosine scoring

For term $t$ in item $i$, with $\mathrm{tf}_{t,i}$ its count and $\mathrm{df}_t$ the number of the
$N$ items containing it:

$$
w_{t,i} = \left(1 + \log \mathrm{tf}_{t,i}\right)\left(1 + \log\frac{1 + N}{1 + \mathrm{df}_t}\right),
\qquad \mathbf{f}_i = \frac{\mathbf{w}_i}{\lVert \mathbf{w}_i \rVert_2}
$$

IDF down-weights words that appear everywhere ("film", "story"). The profile is a confidence-weighted
mean of consumed items and the score is cosine similarity — Rocchio relevance feedback from
information retrieval, where negative feedback, if you have it, enters with negative weight:

$$
\mathbf{p}_u = \frac{\sum_{i \in I_u} c_{ui}\,\mathbf{f}_i}{\sum_{i \in I_u} c_{ui}}, \qquad
\hat{s}_{ui} = \frac{\mathbf{p}_u \cdot \mathbf{f}_i}{\lVert \mathbf{p}_u \rVert\,\lVert \mathbf{f}_i \rVert}
$$

```python
from sklearn.feature_extraction.text import TfidfVectorizer

tfidf = TfidfVectorizer(sublinear_tf=True)        # 1 + log tf, smoothed idf, L2-normalised rows
Fd = tfidf.fit_transform(text).toarray()
vocab = tfidf.get_feature_names_out()
print(f"TF-IDF matrix {Fd.shape}; item 0's heaviest terms: "
      + ", ".join(f"{vocab[j]} ({Fd[0, j]:.2f})" for j in np.argsort(-Fd[0])[:5]))

def content_profile(items):
    p = Fd[list(items)].mean(axis=0)
    return p / (np.linalg.norm(p) + 1e-12)

profiles = {u: content_profile(s) for u, s in seen.items()}
content = lambda u: Fd @ profiles[u] if u in profiles else np.zeros(n_items)
for t in ("warm", "cold-item"):
    print(f"content-based recall@10, {t:>9}: {recall_at_k(content, t):.4f}")
```

```output
TF-IDF matrix (800, 95); item 0's heaviest terms: home (0.52), dark (0.27), dog (0.26), blood (0.26), family (0.25)
content-based recall@10,      warm: 0.0587
content-based recall@10, cold-item: 0.2005
```

On **new items** the content model is the first method that works at all — about three times
random. On **warm** items it loses badly to plain popularity: cosine similarity to a profile knows
nothing about how good or popular an item is, and a mediocre noir film has the same description as
an excellent one.

### What the content model gets wrong: overspecialisation

```python
warm_truth = TASKS["warm"][0]
eval_users = [u for u in warm_truth if u in seen][:600]
fav = {u: np.bincount(genre[list(seen[u])], minlength=K).argmax() for u in eval_users}

def concentration(score_fn, k=10):
    shares, covered = [], set()
    for u in eval_users:
        s = np.asarray(score_fn(u), float)[warm_items].copy()
        s[np.isin(warm_items, list(seen[u]))] = -np.inf
        top = warm_items[np.argpartition(-s, k)[:k]]
        covered.update(top)
        shares.append(np.mean(genre[top] == fav[u]))
    return np.mean(shares), len(covered) / len(warm_items)

actual = np.mean([np.mean(genre[list(warm_truth[u])] == fav[u]) for u in eval_users])
print(f"{'':>12} {'top-10 share in favourite genre':>32} {'catalogue coverage':>19}")
print(f"{'held-out':>12} {actual:>32.3f}")
for name, fn in (("content", content), ("popularity", lambda u: pop)):
    share, cov = concentration(fn)
    print(f"{name:>12} {share:>32.3f} {cov:>19.3f}")
```

```output
              top-10 share in favourite genre  catalogue coverage
    held-out                            0.301
     content                            0.656               0.891
  popularity                            0.140               0.050
```

Users spend 30% of their future interactions in their favourite genre; the content model fills two
thirds of the list with it. That is **overspecialisation**: similarity to the past is the only
signal, so the model recommends more of the same and never makes the jump collaborative filtering
gets for free ("people who liked your noir films also liked this documentary"). The coverage column
is the flip side: content lists are not popularity-biased at all, which is why content features are
the standard tool for the long tail.

## Collaborative filtering on the same split

```python
def implicit_als(X, d=16, alpha=5.0, lam=30.0, n_iter=10, seed=0):
    """Weighted ALS from ch. 5."""
    r = np.random.default_rng(seed)
    U, V = r.normal(0, 0.01, (X.shape[0], d)), r.normal(0, 0.01, (X.shape[1], d))
    Xc, Xt, eye = X.tocsr(), X.T.tocsr(), lam * np.eye(d)
    for _ in range(n_iter):
        for M, Fm, O in ((Xc, U, V), (Xt, V, U)):
            G = O.T @ O + eye
            for row in range(M.shape[0]):
                Oi = O[M.indices[M.indptr[row]:M.indptr[row + 1]]]
                Fm[row] = (np.linalg.solve(G + alpha * Oi.T @ Oi, (1 + alpha) * Oi.sum(axis=0))
                           if len(Oi) else 0)
    return U, V

U_mf, V_mf = implicit_als(X)
mf = lambda u: V_mf @ U_mf[u]
print(f"iALS recall@10: warm {recall_at_k(mf, 'warm'):.4f} | "
      f"cold-item {recall_at_k(mf, 'cold-item'):.4f} (cold item vectors are all zero)")

print(f"\n{'train interactions':>19} {'users':>6} {'seg-pop':>8} {'iALS':>8} {'content':>8}")
for lo, hi in ((0, 3), (3, 6), (6, 12), (12, 25), (25, 10_000)):
    us = [u for u in warm_truth if lo <= n_u[u] < hi]
    cells = [recall_at_k(f, "warm", users=us) for f in (baselines["segment popularity"], mf, content)]
    label = f"{lo}-{hi - 1}" if hi < 10_000 else f"{lo}+"
    print(f"{label:>19} {len(us):>6} " + " ".join(f"{v:>8.4f}" for v in cells))
```

iALS wins the warm task and is no better than random on cold items: the solver never touched those
columns, every cold item scores exactly zero, and tie-breaking picks the top ten. The support
breakdown is the more useful picture. Below three training interactions the iALS user vector is
mostly regularisation and segment popularity beats it outright; from three upward iALS takes over,
and the gap widens with history. A recommender that serves everyone with one model is wrong for
somebody — the whole motivation for hybrids.

## A taxonomy of hybrids

Burke (2002) classified the ways to combine recommenders. Production systems are almost always one
of these, often several stacked:

| Hybrid | How it combines | Example |
|---|---|---|
| Weighted | blend several models' scores | $a\,z(\text{CF}) + (1-a)\,z(\text{content})$ |
| Switching | pick one model per request by a rule | segment popularity for users with < 3 interactions |
| Mixed | show several models' lists side by side | a "new arrivals" row beside "for you" |
| Feature combination | one model consumes both kinds of signal | LightFM, FMs with side features (ch. 6) |
| Feature augmentation | one model's output is another's input feature | CF similarity as a GBDT ranker feature |
| Cascade | a cheap model filters, a precise one re-ranks | retrieval then ranking (ch. 10) |
| Meta-level | one model's learned *model* is another's input | content regressed onto CF factors (below) |

### Weighted and switching, measured

Scores live on different scales — cosine in $[-1, 1]$, iALS dot products near $[0, 1]$,
popularity in the hundreds — so a weighted hybrid z-scores each model per user before blending. A
switching hybrid needs only a rule, and the support table suggests one.

```python
z = lambda s: (s - s.mean()) / (s.std() + 1e-12)
print("weighted: a * z(iALS) + (1 - a) * z(content), warm task")
for a in (0.0, 0.5, 0.8, 1.0):
    print(f"  a = {a:.1f}: recall@10 "
          f"{recall_at_k(lambda u: a * z(mf(u)) + (1 - a) * z(content(u)), 'warm'):.4f}")
print("\nswitching: segment popularity if n_u < threshold else iALS, warm task")
for th in (0, 3, 8):
    switch = lambda u, th=th: seg_pop[segment[u]] if n_u[u] < th else mf(u)
    print(f"  threshold {th}: recall@10 {recall_at_k(switch, 'warm'):.4f}")
```

Neither moves the warm number much, and the reason is instructive. The weighted blend adds nothing
because the content signal is **already inside** the iALS factors: item vectors were learned from
users whose taste was itself shaped by the topics, and blending a model with a noisier copy of its
own information cannot help. The switching rule at three interactions helps a little, only for the
0–2 band, which is a small slice of users — exactly the size of the gain. Hybrids earn their keep where one component is
blind, not where both see the same thing.

## Mapping content into the latent space

The cleanest bridge keeps the CF model for warm items and **learns a function from content to CF
factors**, applied to new items. Gantner et al. (2010) called it attribute-to-feature mapping; van
den Oord et al. (2013) did it with a CNN from audio to a music recommender's factors. In Burke's
terms it is a meta-level hybrid.

### Attribute-to-factor regression

With $\mathbf{F}_W \in \mathbb{R}^{n_W \times p}$ the content of warm items and
$\mathbf{V}_W \in \mathbb{R}^{n_W \times k}$ their learned factors, ridge regression gives

$$
\min_{\mathbf{B}} \lVert \mathbf{F}_W \mathbf{B} - \mathbf{V}_W \rVert_F^2 + \alpha \lVert \mathbf{B} \rVert_F^2
\quad\Longrightarrow\quad
\mathbf{B} = \left(\mathbf{F}_W^\top \mathbf{F}_W + \alpha I\right)^{-1} \mathbf{F}_W^\top \mathbf{V}_W
$$

A cold item gets $\hat{\mathbf{v}}_j = \mathbf{B}^\top \mathbf{f}_j$, scored against the *existing*
user factors, $\hat{s}_{uj} = \mathbf{u}_u \cdot \hat{\mathbf{v}}_j$. Users keep all their
collaborative knowledge; only the item side is imputed.

```python
from sklearn.linear_model import Ridge

for alpha in (0.1, 1.0, 10.0, 30.0, 100.0):
    a2f = Ridge(alpha=alpha).fit(Fd[warm_items], V_mf[warm_items])
    V_hat = a2f.predict(Fd)
    print(f"alpha {alpha:>5}: train R^2 {a2f.score(Fd[warm_items], V_mf[warm_items]):.3f} | "
          f"cold-item recall@10 {recall_at_k(lambda u: V_hat @ U_mf[u], 'cold-item'):.4f}")
V_hat = Ridge(alpha=30.0).fit(Fd[warm_items], V_mf[warm_items]).predict(Fd)
```

```output

alpha   1.0: train R^2 0.345 | cold-item recall@10 0.2026
alpha  10.0: train R^2 0.270 | cold-item recall@10 0.2232

```

Content-to-factor mapping beats the content profile on new items (0.227 against 0.200), because it
pairs the item's content with the user's **collaborative** vector rather than a bag-of-words
profile. Read the columns together: as $\alpha$ grows the training $R^2$ falls from 0.35 to 0.09,
yet ranking quality *rises* and then plateaus. Factors of rarely-seen warm items are noisy targets,
and heavy shrinkage stops the regression chasing them — the best fit is not the best recommender.
Even unregularised, text explains about a third of the factor variance; the rest is the generator's
residual, and no model recovers it from descriptions. For items with *some* data, shrink instead:
$\mathbf{v}_i = \frac{n_i}{n_i + c}\mathbf{v}_i^{\text{CF}} + \frac{c}{n_i + c}\hat{\mathbf{v}}_i$,
the posterior mean under a Gaussian prior centred on the content prediction.

## Feature-sum embeddings and ID dropout

LightFM (Kula, 2015) fuses the two stages. Every user and item is a **set of features**, its vector
is the sum of its features' vectors, and the whole thing trains end to end with a ranking loss:

$$
\mathbf{q}_i = \sum_{f \in \mathcal{F}(i)} x_{if}\,\mathbf{e}_f, \qquad
\mathbf{p}_u = \sum_{g \in \mathcal{G}(u)} x_{ug}\,\mathbf{e}_g, \qquad
\hat{s}_{ui} = \mathbf{p}_u \cdot \mathbf{q}_i + \sum_{f \in \mathcal{F}(i)} x_{if}\,b_f
$$

An indicator for the item's own ID gives a warm item a free per-item correction; a cold item has no
ID feature, so its vector is the sum of its content embeddings. It is the FM of ch. 6 restricted to
user-field-times-item-field interactions. The catch: trained with IDs always present, the model can
push the signal into the ID embeddings and leave the content embeddings undertrained. **DropoutNet**
(Volkovs et al., 2017) randomly zeroes the ID part of the input during training, so the model must
score from features alone on a fraction of examples — the exact condition it faces at serving time.

```python
import torch
import torch.nn as nn

torch.set_num_threads(1)                          # tiny model: one thread beats four here

class FeatureSum(nn.Module):
    """LightFM-style: item = id + TF-IDF words, user = id + segment. Vectors are sums."""
    def __init__(self, d=16):
        super().__init__()
        self.register_buffer("words", torch.as_tensor(Fd, dtype=torch.float32))
        self.register_buffer("seg", torch.as_tensor(segment))
        self.item_id, self.user_id = nn.Embedding(n_items, d), nn.Embedding(n_users, d)
        self.user_seg, self.item_id_b = nn.Embedding(n_segments, d), nn.Embedding(n_items, 1)
        self.word_emb = nn.Parameter(torch.randn(Fd.shape[1], d) * 0.05)
        self.word_b = nn.Parameter(torch.zeros(Fd.shape[1]))
        for e in (self.item_id, self.user_id, self.user_seg):
            nn.init.normal_(e.weight, std=0.05)
        nn.init.zeros_(self.item_id_b.weight)

    def item(self, i, keep):                      # keep = 0 hides the id feature
        return (keep[:, None] * self.item_id(i) + self.words[i] @ self.word_emb,
                keep * self.item_id_b(i)[:, 0] + self.words[i] @ self.word_b)

    def user(self, u, keep):
        return keep[:, None] * self.user_id(u) + self.user_seg(self.seg[u])

def train_feature_sum(p_drop=0.0, epochs=30, batch=1_024, lr=0.01, l2=0.01, seed=0):
    torch.manual_seed(seed)
    g = torch.Generator().manual_seed(seed)
    m = FeatureSum()
    opt = torch.optim.Adam(m.parameters(), lr=lr)
    u_all, i_all = torch.tensor(train["user"].values), torch.tensor(train["item"].values)
    warm_t = torch.tensor(warm_items)
    for _ in range(epochs):
        perm = torch.randperm(len(u_all), generator=g)
        for s in range(0, len(perm), batch):
            b = perm[s:s + batch]
            j = warm_t[torch.randint(0, len(warm_t), (len(b),), generator=g)]
            ku = (torch.rand(len(b), generator=g) >= p_drop).float()   # ID dropout
            ki = (torch.rand(len(b), generator=g) >= p_drop).float()
            pu = m.user(u_all[b], ku)
            (qi, bi), (qj, bj) = m.item(i_all[b], ki), m.item(j, ki)
            reg = (pu.pow(2).sum(1) + qi.pow(2).sum(1) + qj.pow(2).sum(1)).mean()
            loss = -nn.functional.logsigmoid((pu * (qi - qj)).sum(1) + bi - bj).mean() + l2 * reg
            opt.zero_grad()
            loss.backward()
            opt.step()                                                 # BPR, as in ch. 5
    return m

@torch.no_grad()
def feature_sum_scores(m):
    Q, bq = m.item(torch.arange(n_items), torch.as_tensor(~is_cold_item, dtype=torch.float32))
    P = m.user(torch.arange(n_users), torch.as_tensor(~is_cold_user, dtype=torch.float32))
    return (P @ Q.T + bq).numpy()

S_fs = {}
print(f"{'ID dropout':>11} " + " ".join(f"{t:>10}" for t in TASKS))
for p in (0.0, 0.25, 0.5):
    S_fs[p] = feature_sum_scores(train_feature_sum(p_drop=p))
    print(f"{p:>11.2f} " + " ".join(f"{recall_at_k(lambda u: S_fs[p][u], t):>10.4f}" for t in TASKS))
```

## New users: segments, questionnaires, and the hand-off to bandits

A new user offers three sources of information, in rising cost: **attributes** (segment popularity,
or the segment embedding above), an **onboarding questionnaire** ("pick titles you like"), and
**exploration** in the live product. The questionnaire is the interesting one, because you choose
the questions.

### Folding a new user into the factor space

A new user who said yes to items $A$ gets the iALS user solve of ch. 5, shrunk toward the segment's
mean user vector $\boldsymbol{\mu}_s$ instead of toward zero:

$$
\mathbf{x}_u = \underbrace{\left(V^\top V + \alpha V_A^\top V_A + \lambda I\right)^{-1}}_{\propto\ \Sigma_u}
\left((1+\alpha)\,V_A^\top \mathbf{1} + \lambda \boldsymbol{\mu}_s\right)
$$

With no answers this is the segment's average taste; each yes pulls it toward the liked items. The
inverted matrix is a posterior covariance $\Sigma_u$, and choosing questions that shrink it fastest
is **D-optimal design**: greedily ask about the item maximising $\mathbf{v}_j^\top \Sigma
\mathbf{v}_j$, the variance of the prediction you are least sure of. Asking adaptively — re-solving
after each answer and picking by estimate plus uncertainty — is a bandit.

```python
cu_truth = TASKS["cold-user"][0]
aff = user_vec @ item_vec.T / 2.0                 # the oracle a questionnaire answer comes from
G, d_f = V_mf.T @ V_mf, V_mf.shape[1]
seg_mu = np.array([U_mf[warm_users[segment[warm_users] == s]].mean(axis=0) for s in range(n_segments)])
POOL = warm_items[np.argsort(-pop[warm_items])][:60]      # recognisable titles to ask about
CAND = np.setdiff1d(warm_items, POOL)                     # ranked and scored without the pool

def fold_in(liked, mu, alpha=5.0, lam=300.0):
    Va = V_mf[liked].reshape(-1, d_f)
    A = G + alpha * Va.T @ Va + lam * np.eye(d_f)
    return np.linalg.solve(A, (1 + alpha) * Va.sum(axis=0) + lam * mu), np.linalg.inv(A)

def pred_var(Sigma):
    return np.einsum("ij,jk,ik->i", V_mf[POOL], Sigma, V_mf[POOL])

def d_optimal(m):
    A, chosen = np.eye(d_f), []
    for _ in range(m):
        gain = pred_var(np.linalg.inv(A))
        gain[np.isin(POOL, chosen)] = -np.inf
        chosen.append(POOL[np.argmax(gain)])
        A += np.outer(V_mf[chosen[-1]], V_mf[chosen[-1]])
    return np.array(chosen)

def adaptive(u, m, mu, thr):
    """Ask, update, ask again: estimate + uncertainty, the LinUCB rule of ch. 11."""
    asked, liked = [], []
    for _ in range(m):
        x, Sigma = fold_in(liked, mu)
        ucb = V_mf[POOL] @ x + np.sqrt(pred_var(Sigma))
        ucb[np.isin(POOL, asked)] = -np.inf
        asked.append(POOL[np.argmax(ucb)])
        liked += [asked[-1]] if aff[u, asked[-1]] > thr else []
    return liked

DOPT, rq = d_optimal(30), np.random.default_rng(3)

def onboarding(strategy, m, k=10):
    rec, yes = [], []
    for u, items in cu_truth.items():
        tr = items & set(CAND)
        if not tr:
            continue
        mu, thr = seg_mu[segment[u]], np.quantile(aff[u, warm_items], 0.8)   # yes = top 20% taste
        if strategy == "adaptive":
            liked = adaptive(u, m, mu, thr)
        else:
            asked = {"popular": POOL[:m], "random": rq.choice(POOL, m, replace=False),
                     "d-optimal": DOPT[:m]}[strategy]
            liked = [i for i in asked if aff[u, i] > thr]
        top = CAND[np.argpartition(-(V_mf[CAND] @ fold_in(liked, mu)[0]), k)[:k]]
        rec.append(len(tr & set(top)) / min(k, len(tr)))
        yes.append(len(liked))
    return np.mean(rec), np.mean(yes)

print(f"no questions (segment prior only): recall@10 {onboarding('popular', 0)[0]:.4f}")
print(f"\n{'questions':>9} {'strategy':>10} {'recall@10':>10} {'yes answers':>12}")
for m in (10, 20):
    for strat in ("popular", "random", "d-optimal", "adaptive"):
        rcl, y = onboarding(strat, m)
        print(f"{m:>9} {strat:>10} {rcl:>10.4f} {y:>12.2f}")
```

runs on live impressions: the questionnaire is the first few rounds of a contextual bandit, and the
hand-off is seamless if your fold-in keeps $\Sigma_u$ around.

## Try it Yourself

```python
import numpy as np
from scipy import sparse
from scipy.sparse.linalg import svds
from sklearn.linear_model import Ridge

def content_value(residual, n_users=1_500, n_items=600, d=8, K=8, seed=0):
    """How much does content buy you on new items, as the unexplained part of taste grows?"""
    r = np.random.default_rng(seed)
    theta = r.dirichlet(np.full(K, 0.3), n_items)
    item = 1.6 * theta @ r.normal(0, 1, (K, d)) + r.normal(0, residual, (n_items, d))
    item *= 0.98 / item.std()                                 # total taste variance held fixed
    user = r.normal(0, 1, (n_users, d))
    logit = user @ item.T / 2 - 4.0 + np.log(r.lognormal(-1, 1, n_items))[None, :]
    Y = (r.random((n_users, n_items)) < 1 / (1 + np.exp(-logit))).astype(float)
    cold = r.random(n_items) < 0.2
    U, s, Vt = svds(sparse.csr_matrix(Y[:, ~cold]), k=d)     # CF factors from warm items only
    V_warm = Vt.T * s
    reg = Ridge(alpha=1.0).fit(theta[~cold], V_warm)          # attribute-to-factor
    oracle = np.linalg.lstsq(item[~cold], V_warm, rcond=None)[0]   # if features were the truth
    Yc, idx = Y[:, cold], np.flatnonzero(Y[:, cold].sum(1) > 0)
    def recall(S):
        top = np.argpartition(-S[idx], 10, axis=1)[:, :10]
        hits = np.take_along_axis(Yc[idx], top, axis=1).sum(1)
        return np.mean(hits / np.minimum(10, Yc[idx].sum(1)))
    return (reg.score(theta[~cold], V_warm), recall(U @ reg.predict(theta[cold]).T),
            recall(U @ (item[cold] @ oracle).T), recall(r.random((n_users, int(cold.sum())))))

print(f"{'residual sd':>12} {'R^2':>6} {'attr->factor':>13} {'oracle':>7} {'random':>7}")
for residual in (0.1, 0.55, 1.0, 2.0, 4.0):
    r2, a, o, b = content_value(residual)
    print(f"{residual:>12.2f} {r2:>6.3f} {a:>13.4f} {o:>7.4f} {b:>7.4f}")
```

move it.

```python
def few_shot(n_reveal, k=10, seed=0):
    r = np.random.default_rng(seed)
    res = {"segment pop": [], "content": [], "fold-in": []}
    for u, items in cu_truth.items():
        items = np.array(sorted(items))
        if len(items) <= n_reveal:
            continue
        revealed = r.choice(items, n_reveal, replace=False)
        truth = set(items) - set(revealed)
        cand = np.setdiff1d(warm_items, revealed)
        x, _ = fold_in(revealed, seg_mu[segment[u]], lam=30.0)
        scores = {"segment pop": seg_pop[segment[u]][cand],
                  "content": Fd[cand] @ content_profile(revealed),
                  "fold-in": V_mf[cand] @ x}
        for name, s in scores.items():
            top = cand[np.argpartition(-s, k)[:k]]
            res[name].append(len(truth & set(top)) / min(k, len(truth)))
    return {name: np.mean(v) for name, v in res.items()}, len(res["content"])

    print(f"{n:>8} {n_eval:>6} {out['segment pop']:>12.4f} {out['content']:>8.4f} "
          f"{out['fold-in']:>8.4f}")
```
