## Order is information

Someone who just bought a camera is about to want a memory card, not a second camera. Matrix
factorisation (ch. 4–5) treats each user as a bag of items and asks what they like *overall*;
sequential recommendation asks what they want *next*, given the order of what they just did.

The generator below makes order carry signal. Every item has three natural successors (an episode
and the next one, a phone and its case). Each step follows the last item 30% of the time, the item
*two* back 15% (a dependency no first-order model can see), stays on the session's current genre —
its *intent* — 45%, and otherwise falls back to popularity.

```python
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.nn.functional as F
from scipy import sparse

torch.set_num_threads(1)
torch.manual_seed(0)
rng = np.random.default_rng(0)
n_users, n_items, n_genres = 3_000, 500, 8
genre = rng.integers(0, n_genres, n_items)
appeal = rng.lognormal(0, 1.2, n_items)

succ = np.zeros((n_items, 3), dtype=int)        # each item has 3 natural "next" items
for i in range(n_items):
    same = np.flatnonzero((genre == genre[i]) & (np.arange(n_items) != i))
    succ[i] = rng.choice(same, 3, replace=False, p=appeal[same] / appeal[same].sum())
    if rng.random() < 0.3:
        succ[i, 2] = rng.integers(0, n_items)   # some links cross genres

def draw_from_genre(g, r):
    c = np.flatnonzero(genre == g)
    return r.choice(c, p=appeal[c] / appeal[c].sum())

def make_sequences(n, p_follow=0.30, p_skip=0.15, p_intent=0.45, seed=0):
    r = np.random.default_rng(seed)
    out = []
    for _ in range(n):
        length = min(8 + r.geometric(1 / 18), 50)
        intent = r.integers(n_genres)
        s = [draw_from_genre(intent, r)]
        while len(s) < length:
            if r.random() < 0.05:
                intent = r.integers(n_genres)                  # the user changes their mind
            x = r.random()
            if x < p_follow:                                    # follow the last item
                nxt = succ[s[-1], r.choice(3, p=[0.5, 0.3, 0.2])]
            elif x < p_follow + p_skip and len(s) >= 2:         # follow the item two back
                nxt = succ[s[-2], 0]
            elif x < p_follow + p_skip + p_intent:              # stay on the session's intent
                nxt = draw_from_genre(intent, r)
            else:                                               # global popularity
                nxt = r.choice(n_items, p=appeal / appeal.sum())
            s.append(int(nxt))
        out.append(np.array(s))
    return out

seqs = make_sequences(n_users)
lengths = np.array([len(s) for s in seqs])
pairs = np.concatenate([np.stack([s[:-1], s[1:]], 1) for s in seqs])

def entropy_bits(counts):
    p = counts[counts > 0] / counts.sum()
    return float(-(p * np.log2(p)).sum())

h_next = entropy_bits(np.bincount(pairs[:, 1], minlength=n_items))
h_joint = entropy_bits(np.bincount(pairs[:, 0] * n_items + pairs[:, 1]))
h_prev = entropy_bits(np.bincount(pairs[:, 0], minlength=n_items))
print(f"{len(seqs):,} sequences | {lengths.sum():,} events | "
      f"length median {np.median(lengths):.0f}, max {lengths.max()}")
print(f"H(next item)              = {h_next:.2f} bits")
print(f"H(next item | last item)  = {h_joint - h_prev:.2f} bits  (plug-in estimate, biased low)")
print(f"share of steps that go to one of the last item's 3 successors: "
      f"{np.mean([b in succ[a] for a, b in pairs]):.3f}")
```

```output
3,000 sequences | 73,853 events | length median 21, max 50
H(next item)              = 7.98 bits
H(next item | last item)  = 5.09 bits  (plug-in estimate, biased low)
share of steps that go to one of the last item's 3 successors: 0.361
```

Knowing the last item cuts the uncertainty about the next one from 7.98 to about 5.09 bits — from
"one of 250 equally likely items" to "one of 34". The plug-in estimate is biased low (250,000 cells,
71k pairs), so read the 2.9-bit gap as an upper bound. 36.1% of steps land on a successor of the
last item: the injected 30% plus intent and popularity draws that happen to.

## Leave-last-out, and what it does not protect you from

Sequential papers evaluate by **leave-last-out** (ch. 2): each user's last event is the test target,
the second-to-last validates, everything earlier trains. The model ranks the full catalogue of 500
items for each test case — no sampled negatives, for reasons the final table makes concrete.

```python
train_seqs = [s[:-2] for s in seqs]                  # everything except the last two events
valid = [(s[:-2], s[-2]) for s in seqs]              # predict the second-to-last
test = [(s[:-1], s[-1]) for s in seqs]               # predict the last
item_count = np.bincount(np.concatenate(train_seqs), minlength=n_items).astype(float)

def ranks_of_target(score_fn, cases, chunk=1_000):
    out = []
    for st in range(0, len(cases), chunk):
        part = cases[st:st + chunk]
        S = np.asarray(score_fn([c[0] for c in part], st), dtype=np.float64)
        t = S[np.arange(len(part)), [c[1] for c in part]]
        out.append((S > t[:, None]).sum(1) + 1)          # rank of the target in the full catalogue
    return np.concatenate(out)

def metrics(ranks, k=10):
    hit = ranks <= k
    return float(hit.mean()), float(np.where(hit, 1 / np.log2(ranks + 1), 0.0).mean())

popularity = lambda prefixes, offset: np.tile(item_count, (len(prefixes), 1))

results = {}
results["popularity"] = ranks_of_target(popularity, test)
r, n = metrics(results["popularity"])
print(f"train events {sum(map(len, train_seqs)):,} | test cases {len(test):,} (one per user)")
print(f"popularity: recall@10 {r:.4f} | ndcg@10 {n:.4f}")
print(f"test target already in the user's history: {np.mean([t in set(p) for p, t in test]):.3f}")
```

```output
train events 67,853 | test cases 3,000 (one per user)
popularity: recall@10 0.1713 | ndcg@10 0.0858
test target already in the user's history: 0.314
```

Popularity puts the true next item in the top 10 for 17.1% of users: the floor. And 31.4% of targets
are already in the user's history, so unlike ch. 2–7 we do **not** filter seen items —
re-consumption (a grocery staple, a favourite song) is part of the sequential task.

Leave-last-out has a blind spot this data cannot show: no global clock. In real logs user A's last
event may be in March while user B's training runs into June, so the future leaks in (Ji et al.,
2023). A global time cut (ch. 2) is production-faithful; if you report leave-last-out, say so.

## Markov chains: the next item depends on the last one

The simplest sequence model forgets all but the last item: count how often $j$ followed $i$ and
recommend the most frequent followers. A lookup table — and on real data, embarrassingly hard to beat.

### The smoothed transition estimate

This shows that "add-$\beta$" smoothing is the Bayesian posterior mean, not a hack. Let $C_{ij}$ be
the number of times $j$ followed $i$ in training and $\pi_j$ the overall popularity share of $j$. Put a
Dirichlet prior $\mathrm{Dir}(\beta\boldsymbol{\pi})$ on the row $P(\cdot \mid i)$; the counts are
multinomial, so the posterior is $\mathrm{Dir}(\beta\boldsymbol{\pi} + C_{i\cdot})$ with mean

$$
\hat{P}(j \mid i) = \frac{C_{ij} + \beta\,\pi_j}{\sum_{j'} C_{ij'} + \beta}
$$

A rarely seen last item falls back to popularity; a frequent one trusts its counts; $\beta$ is the
prior's pseudo-count. The gap-2 variant adds a table of pairs two steps apart.

```python
def transition_counts(seqs_, gap=1):
    T = np.zeros((n_items, n_items))
    for s in seqs_:
        if len(s) > gap:
            np.add.at(T, (s[:-gap], s[gap:]), 1.0)
    return T

prior = item_count / item_count.sum()
T1, T2 = transition_counts(train_seqs, 1), transition_counts(train_seqs, 2)

def smoothed(T, items, beta=1.0):
    C = T[items]
    return (C + beta * prior) / (C.sum(1, keepdims=True) + beta)

markov = lambda prefixes, offset: smoothed(T1, [p[-1] for p in prefixes])

def markov_two_back(prefixes, offset, w=0.5):
    last2 = [p[-2] if len(p) >= 2 else p[-1] for p in prefixes]
    return smoothed(T1, [p[-1] for p in prefixes]) + w * smoothed(T2, last2)

shuffled = [np.random.default_rng(i).permutation(s) for i, s in enumerate(train_seqs)]
T_shuf = transition_counts(shuffled, 1)
markov_shuffled = lambda prefixes, offset: smoothed(T_shuf, [p[-1] for p in prefixes])

for name, fn in (("Markov, shuffled training order", markov_shuffled),
                 ("Markov (first order)", markov),
                 ("Markov + gap-2 rules", markov_two_back)):
    results[name] = ranks_of_target(fn, test)
    r, n = metrics(results[name])
    print(f"{name:>32}: recall@10 {r:.4f} | ndcg@10 {n:.4f}")
```

```output
 Markov, shuffled training order: recall@10 0.4370 | ndcg@10 0.2484
            Markov (first order): recall@10 0.5583 | ndcg@10 0.3552
            Markov + gap-2 rules: recall@10 0.6307 | ndcg@10 0.3964
```

Shuffling each training sequence costs 0.121 recall (0.558 to 0.437); what survives is same-genre
co-occurrence. The ordered chain is 3.3x popularity.
The gap-2 table — a crude *sequential rule* (Ludewig and Jannach, 2018) — adds another 0.072 because
it counts the generator's skip dependency directly. Remember 0.6307; it comes back at the end.

### FPMC: factorising the transition matrix

A transition matrix has $n^2$ cells — $10^{12}$ for a million items — so it cannot be estimated at
scale. FPMC (Rendle et al., 2010) factorises it and adds an MF term for long-term taste; for user $u$
with previous basket $B_\ell$, item $j$ scores

$$
\hat{x}_{u,\ell,j} = \underbrace{\mathbf{v}^{UI}_u \cdot \mathbf{v}^{IU}_j}_{\text{MF: long-term taste}}
+ \underbrace{\frac{1}{|B_{\ell}|}\sum_{i \in B_{\ell}} \mathbf{v}^{IL}_j \cdot \mathbf{v}^{LI}_i}_{\text{factorised Markov chain}}
$$

A taste pair ($\mathbf{v}^{UI}, \mathbf{v}^{IU}$) and a transition pair ($\mathbf{v}^{IL}, \mathbf{v}^{LI}$)
make transitions low-rank, so similar items share strength; training is BPR (ch. 5). The neural
models below replace "the last basket" with a learned summary of the prefix, same dot-product top.

## Session-kNN: find sessions that look like this one

Instead of modelling transitions, find training sessions that look like the current one and vote
with their items: user-kNN (ch. 3) with the session as the "user" and recency weights on its
items. On public session datasets, tuned session-kNN matched or beat GRU4Rec in most comparisons
(Ludewig and Jannach, 2018; Ludewig et al., 2021) — a famous case of the neural model not winning.

```python
rows = np.concatenate([np.full(len(s), i) for i, s in enumerate(train_seqs)])
S_mat = sparse.csr_matrix((np.ones(len(rows)), (rows, np.concatenate(train_seqs))),
                          shape=(len(train_seqs), n_items))
S_mat.data[:] = 1.0                                   # binary session-item matrix
sess_norm = np.sqrt(np.asarray(S_mat.sum(1)).ravel())

def session_knn(prefixes, offset, k_nn=100, m=5):
    Q = np.zeros((len(prefixes), n_items))
    for r_, p in enumerate(prefixes):
        recent = p[-m:]
        np.add.at(Q[r_], recent, np.arange(1, len(recent) + 1) / len(recent))  # recency weights
    sims = np.asarray((S_mat @ Q.T).T) / (sess_norm[None, :] + 1e-9)
    own = np.arange(offset, offset + len(prefixes))
    sims[np.arange(len(prefixes)), own] = 0.0          # never match the user's own training prefix
    kth = -np.partition(-sims, k_nn, axis=1)[:, [k_nn]]
    sims = np.where(sims > kth, sims, 0.0)
    scores = np.asarray((sparse.csr_matrix(sims) @ S_mat).todense())
    return scores + 1e-6 * prior                       # popularity breaks ties

for m in (1, 2, 5, 10):
    r, n = metrics(ranks_of_target(lambda P, o: session_knn(P, o, m=m), valid))
    print(f"session-kNN, last {m:>2} items (validation): recall@10 {r:.4f} | ndcg@10 {n:.4f}")
results["session-kNN"] = ranks_of_target(lambda P, o: session_knn(P, o, m=2), test)
r, n = metrics(results["session-kNN"])
print(f"session-kNN, m=2 (test): recall@10 {r:.4f} | ndcg@10 {n:.4f}")
```

```output
session-kNN, last  1 items (validation): recall@10 0.5137 | ndcg@10 0.2560
session-kNN, last  2 items (validation): recall@10 0.5190 | ndcg@10 0.2560
session-kNN, last  5 items (validation): recall@10 0.4947 | ndcg@10 0.2471
session-kNN, last 10 items (validation): recall@10 0.4490 | ndcg@10 0.2332
session-kNN, m=2 (test): recall@10 0.5053 | ndcg@10 0.2554
```

Validation picks the last $m = 2$ items; longer windows dilute as intent drifts. On test, session-kNN
trails first-order Markov (recall 0.505 against 0.558, NDCG 0.255 against 0.355). That is this
generator's bias: session-kNN scores what a neighbour *contained*, not what came *right after* the
match. Real click logs are often more intent-driven, and V-SKNN/STAN add position awareness.

## GRU4Rec: a running summary of the session

GRU4Rec (Hidasi et al., 2016) reads the session one item at a time and keeps a running summary, the
hidden state $\mathbf{h}_t = \mathrm{GRU}(\mathbf{h}_{t-1}, \mathbf{e}_{x_t})$ ($\mathbf{e}_{x_t}$ is the
embedding of the item at step $t$), and predicts the next item at every step with a softmax,
$P(x_{t+1} = j) \propto \exp(\mathbf{w}_j \cdot \mathbf{h}_t + b_j)$. The state can remember two steps back.

### Session-parallel mini-batches

Sessions differ wildly in length and padding wastes compute, so GRU4Rec's loader runs $B$ parallel
*lanes*: each advances one session a step per batch; when a session ends, the next in the queue takes
over the lane with its hidden state reset to zero. With `[10, 11, 12]`,
`[20, 21]`, `[30, 31, 32]` in three lanes, step 0 feeds `[10, 20, 30]`; lane 2 is then exhausted, so
step 1 feeds `[11, 40, 31]` with only lane 2 reset. Our sequences are short, so the code below simply
left-pads to the last 30 items with a next-item loss at every real position — same objective.

```python
MAXLEN, PAD = 30, n_items

def pad_left(seq_list):
    X = np.full((len(seq_list), MAXLEN), PAD, dtype=np.int64)
    for r_, s in enumerate(seq_list):
        s = s[-MAXLEN:]
        X[r_, MAXLEN - len(s):] = s
    return torch.from_numpy(X)

inp = pad_left([s[:-1] for s in train_seqs])            # x_1 .. x_{n-1}
tgt = pad_left([s[1:] for s in train_seqs])             # x_2 .. x_n: a target at every position

class GRU4Rec(nn.Module):
    def __init__(self, d=64, drop=0.2):
        super().__init__()
        self.emb = nn.Embedding(n_items + 1, d, padding_idx=PAD)
        self.gru = nn.GRU(d, d, batch_first=True)
        self.drop = nn.Dropout(drop)
        self.out = nn.Linear(d, n_items)

    def forward(self, x):                               # (B, L) -> hidden states (B, L, d)
        h, _ = self.gru(self.drop(self.emb(x)))
        return self.drop(h)

    def logits(self, h):
        return self.out(h)

def full_softmax_loss(model, h, y):
    return F.cross_entropy(model.logits(h), y)

def train_seq_model(model, loss_fn=full_softmax_loss, epochs=16, lr=3e-3, batch=128, seed=0):
    torch.manual_seed(seed)
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    g = torch.Generator().manual_seed(seed)
    for epoch in range(epochs):
        model.train()
        perm = torch.randperm(len(inp), generator=g)
        for st in range(0, len(perm), batch):
            b = perm[st:st + batch]
            y = tgt[b]
            keep = y != PAD                              # only real positions carry a loss
            loss = loss_fn(model, model(inp[b])[keep], y[keep])
            opt.zero_grad(); loss.backward(); opt.step()
    return model

def scorer(model):
    @torch.no_grad()
    def fn(prefixes, offset):
        model.eval()
        return model.logits(model(pad_left(prefixes))[:, -1]).numpy()
    return fn

def report(name, model):
    for split, cases in (("validation", valid), ("test", test)):
        rk = ranks_of_target(scorer(model), cases)
        print(f"{name} ({split}): recall@10 {metrics(rk)[0]:.4f} | ndcg@10 {metrics(rk)[1]:.4f}")
    return rk

gru = train_seq_model(GRU4Rec())
results["GRU4Rec"] = report("GRU4Rec, full softmax", gru)
```

```output
GRU4Rec, full softmax (validation): recall@10 0.5990 | ndcg@10 0.3640
GRU4Rec, full softmax (test): recall@10 0.5883 | ndcg@10 0.3593
```

GRU4Rec reaches 0.588 recall and 0.359 NDCG: above first-order Markov (0.558 / 0.355), but **below
the gap-2 count table (0.631 / 0.396)**: it learned only part of the skip dependency from 68k events.

## Which loss: full softmax, BPR, TOP1, sampled softmax

Scoring all 500 items at every position is cheap; ten million is not, so the original GRU4Rec scored
a *sample* $S$ of negatives with ranking losses. With $r_+$ the true next item's score and $r_j$ a
sampled negative's:

$$
\mathcal{L}_{\text{CE}} = -\log\frac{e^{r_{+}}}{\sum_{j=1}^{N} e^{r_j}}, \qquad
\mathcal{L}_{\text{BPR}} = -\frac{1}{N_S}\sum_{j \in S}\log\sigma(r_{+} - r_j), \qquad
\mathcal{L}_{\text{TOP1}} = \frac{1}{N_S}\sum_{j \in S}\left[\sigma(r_j - r_{+}) + \sigma(r_j^2)\right]
$$

BPR pushes the positive above each negative; TOP1 adds $\sigma(r_j^2)$ to pull negative scores toward
zero. Sampled softmax keeps cross-entropy over the positive plus $S$, with ch. 7's logQ correction.

```python
log_q = torch.as_tensor(np.log(prior), dtype=torch.float32)
sampler = torch.distributions.Categorical(probs=torch.as_tensor(prior, dtype=torch.float32))

def sampled_losses(kind, n_neg=64):
    def loss_fn(model, h, y):
        neg = sampler.sample((n_neg,))                   # shared, popularity-distributed negatives
        W, b = model.out.weight, model.out.bias
        pos = (h * W[y]).sum(-1) + b[y]
        negs = h @ W[neg].T + b[neg]
        if kind == "bpr":
            return -F.logsigmoid(pos[:, None] - negs).mean()
        if kind == "top1":
            return (torch.sigmoid(negs - pos[:, None]) + torch.sigmoid(negs ** 2)).mean()
        logits = torch.cat([pos[:, None] - log_q[y][:, None], negs - log_q[neg][None]], 1)  # logQ
        return F.cross_entropy(logits, torch.zeros(len(y), dtype=torch.long))
    return loss_fn

print(f"{'loss':>34} {'recall@10':>10} {'ndcg@10':>9}\n{'full softmax over 500 items':>34} "
      f"{metrics(results['GRU4Rec'])[0]:>10.4f} {metrics(results['GRU4Rec'])[1]:>9.4f}")
for kind, label in (("bpr", "BPR, 64 sampled negatives"), ("top1", "TOP1, 64 sampled negatives"),
                    ("softmax", "sampled softmax + logQ, 64 neg")):
    m_ = train_seq_model(GRU4Rec(), loss_fn=sampled_losses(kind), epochs=12)
    r, n = metrics(ranks_of_target(scorer(m_), test))
    print(f"{label:>34} {r:>10.4f} {n:>9.4f}")
```

```output
                              loss  recall@10   ndcg@10
       full softmax over 500 items     0.5883    0.3593
         BPR, 64 sampled negatives     0.3197    0.1819
        TOP1, 64 sampled negatives     0.2263    0.1060
    sampled softmax + logQ, 64 neg     0.5367    0.3183
```

With 64 popularity-sampled negatives, BPR keeps only 54% of the full-softmax recall and TOP1 39%:
once the positive beats the typical easy negative, the sigmoid saturates and the gradient vanishes.
Sampled softmax with logQ recovers most of the gap (0.537 against 0.588). Later GRU4Rec versions
(Hidasi and Karatzoglou, 2018) moved to cross-entropy and BPR-max with thousands of negatives, and
Hidasi and Czapp (2023) found third-party ports getting exactly these details wrong.

## SASRec: attention over the whole history

A GRU squeezes the whole prefix through one vector, step by step. SASRec (Kang and McAuley, 2018)
lets position $t$ look back at *every* earlier item directly and learn which ones matter — the item
two back is one attention weight away instead of surviving two recurrent updates.

### Causal self-attention

This shows what "causal" means and why one pass trains every position. Position $t$'s input
$\mathbf{e}_{x_t} + \mathbf{p}_t$ (item plus learned position embedding) maps linearly to a query
$\mathbf{q}_t$, key $\mathbf{k}_t$ and value $\mathbf{v}_t$; the output averages values at $s \le t$ only:

$$
\mathrm{Attn}(H)_t = \sum_{s \le t} \alpha_{ts}\, \mathbf{v}_s, \qquad
\alpha_{ts} = \frac{\exp(\mathbf{q}_t \cdot \mathbf{k}_s / \sqrt{d})}{\sum_{s' \le t}\exp(\mathbf{q}_t \cdot \mathbf{k}_{s'} / \sqrt{d})}
$$

```python
class CausalBlock(nn.Module):
    def __init__(self, d, heads, drop):
        super().__init__()
        self.n1, self.n2 = nn.LayerNorm(d), nn.LayerNorm(d)
        self.qkv, self.proj = nn.Linear(d, 3 * d), nn.Linear(d, d)
        self.ff = nn.Sequential(nn.Linear(d, 2 * d), nn.GELU(), nn.Linear(2 * d, d))
        self.heads, self.drop = heads, nn.Dropout(drop)

    def forward(self, h, allowed):
        B, L, d = h.shape
        q, k, v = (self.qkv(self.n1(h)).reshape(B, L, 3, self.heads, d // self.heads)
                   .permute(2, 0, 3, 1, 4))
        a = F.scaled_dot_product_attention(q, k, v, attn_mask=allowed[:, None],
                                           dropout_p=self.drop.p if self.training else 0.0)
        h = h + self.drop(self.proj(a.transpose(1, 2).reshape(B, L, d)))
        return h + self.drop(self.ff(self.n2(h)))

class SASRec(nn.Module):
    def __init__(self, d=64, n_layers=1, heads=1, drop=0.1):
        super().__init__()
        self.emb = nn.Embedding(n_items + 1, d, padding_idx=PAD)
        self.pos = nn.Embedding(MAXLEN, d)                 # learned absolute positions
        nn.init.normal_(self.emb.weight, std=0.05); nn.init.normal_(self.pos.weight, std=0.05)
        self.blocks = nn.ModuleList([CausalBlock(d, heads, drop) for _ in range(n_layers)])
        self.norm, self.drop = nn.LayerNorm(d), nn.Dropout(drop)
        self.register_buffer("causal", torch.tril(torch.ones(MAXLEN, MAXLEN, dtype=torch.bool)))
        self.register_buffer("diag", torch.eye(MAXLEN, dtype=torch.bool))

    def forward(self, x):
        allowed = (self.causal[None] & (x != PAD)[:, None, :]) | self.diag[None]
        h = self.drop(self.emb(x) + self.pos.weight[None])
        for blk in self.blocks: h = blk(h, allowed)
        return self.norm(h)

    def logits(self, h):                       # output layer tied to the input embeddings
        return h @ self.emb.weight[:n_items].T

x_demo = pad_left([np.array([7, 3, 9])])
m_demo = SASRec().eval()
with torch.no_grad():
    base = m_demo.logits(m_demo(x_demo))
    x_future = x_demo.clone(); x_future[0, -1] = 123        # change the LAST item only
    changed = m_demo.logits(m_demo(x_future))
diff = (base - changed).abs().amax(-1)[0, -3:]
print(f"max logit change at the last 3 positions after editing the final item: {diff.tolist()}")

sas = train_seq_model(SASRec())
results["SASRec"] = report("SASRec", sas)
```

The first line is the causality test: editing the final item moves logits only at the final
position, so no position sees its future and all 30 per-position losses are legitimate (the diagonal
is always allowed, so padded rows never divide by zero). SASRec reaches 0.571 recall / 0.350 NDCG —
level with GRU4Rec and first-order Markov, below the gap-2 table.

## BERT4Rec: predict hidden items instead of the next one

SASRec reads left to right. BERT4Rec (Sun et al., 2019) hides a random subset $\mathcal{M}$ of
positions behind a `[MASK]` token and predicts them from *both* sides with a bidirectional
transformer — the Cloze task:

$$
\mathcal{L}_{\text{Cloze}} = -\frac{1}{|\mathcal{M}|}\sum_{t \in \mathcal{M}} \log P\big(x_t \mid x_{\setminus \mathcal{M}}\big)
$$

At inference a `[MASK]` is appended after the last item. The reported win over SASRec has not held
up: Petrov and Macdonald (2022) found BERT4Rec hard to reproduce and slow to converge, and Klenitskiy
and Vasilev (2023) showed SASRec with full cross-entropy (not its original one-negative loss)
matches or beats it — the loss, not the bidirectionality. The Exercise measures that gap here.

## The honest comparison

Every model on the same 3,000 test cases, full-catalogue ranking, a bootstrap NDCG interval, and —
last column — what most pre-2020 sequential papers reported: the target ranked among 100 random items.

```python
def sampled_ranks(score_fn, cases, n_neg=100, seed=0):
    r_ = np.random.default_rng(seed)
    out = []
    for st in range(0, len(cases), 1_000):
        part = cases[st:st + 1_000]
        S = np.asarray(score_fn([c[0] for c in part], st), dtype=np.float64)
        for row, (p, t) in enumerate(part):
            neg = r_.choice(np.delete(np.arange(n_items), t), n_neg, replace=False)
            out.append(int((S[row, neg] > S[row, t]).sum()) + 1)
    return np.array(out)

fns = {"popularity": popularity, "Markov (first order)": markov, "Markov + gap-2 rules": markov_two_back,
       "session-kNN": lambda P, o: session_knn(P, o, m=2),
       "GRU4Rec": scorer(gru), "SASRec": scorer(sas)}
boot = np.random.default_rng(1).integers(0, len(test), (300, len(test)))
rows_ = []
for name, fn in fns.items():
    rk = results[name]
    ndcg_u = np.where(rk <= 10, 1 / np.log2(rk + 1), 0.0)
    lo, hi = np.percentile(ndcg_u[boot].mean(1), [2.5, 97.5])
    r, n = metrics(rk)
    rows_.append({"model": name, "recall@10": round(r, 4), "ndcg@10": round(n, 4),
                  "ndcg 95% CI": f"[{lo:.3f}, {hi:.3f}]",
                  "HR@10, 100 sampled neg": round(float((sampled_ranks(fn, test) <= 10).mean()), 4)})
print(pd.DataFrame(rows_).to_string(index=False))
```

```output

          popularity     0.1713   0.0858 [0.079, 0.094]                  0.4423
Markov (first order)     0.5583   0.3552 [0.341, 0.368]                  0.7367
Markov + gap-2 rules     0.6307   0.3964 [0.383, 0.409]                  0.7913
         session-kNN     0.5053   0.2554 [0.245, 0.265]                  0.7837

              SASRec     0.5713   0.3500 [0.337, 0.363]                  0.8093
```

Full ranking first. The gap-2 count table wins outright: NDCG 0.396, interval [0.383, 0.409] clear
of every other model. GRU4Rec (0.359), first-order Markov (0.355) and SASRec (0.350) overlap — one
tier. Session-kNN is a tier below. **The neural models did not automatically win**; a count table
matched to the generator did. On real data nobody hands you the structure — the honest case for
neural models — but that case is won on full-ranking leaderboards against tuned baselines.

Now the last column. Sampled HR@10 says session-kNN (0.784) beats first-order Markov (0.737) — **the
reverse of full ranking** — both neural models (0.807, 0.809) beat the gap-2 table (0.791), and SASRec
edges GRU4Rec: three reversals. This is Krichene and Rendle (2020) in one table: random negatives are
mostly easy, unpopular items, so the test rewards broad plausibility (session-kNN's neighbour votes)
over pinning the exact next item. Sampled metrics do not just compress the real ones; they reorder.

## Try it Yourself

One knob: `p_follow` is the chance the next item is one of the last item's two successors (otherwise
popularity draws it). Watch where the Markov chain overtakes popularity.

```python
import numpy as np

def experiment(p_follow, n_items=300, n_seq=2_000, seed=0):
    r = np.random.default_rng(seed)
    appeal = r.lognormal(0, 1.2, n_items)
    succ = r.integers(0, n_items, (n_items, 2))
    seqs = []
    for _ in range(n_seq):
        s = [r.integers(n_items)]
        for _ in range(r.integers(5, 30)):
            if r.random() < p_follow:
                s.append(succ[s[-1], r.integers(2)])
            else:
                s.append(r.choice(n_items, p=appeal / appeal.sum()))
        seqs.append(np.array(s))
    train = [s[:-1] for s in seqs]
    count = np.bincount(np.concatenate(train), minlength=n_items).astype(float)
    T = np.zeros((n_items, n_items))
    for s in train:
        np.add.at(T, (s[:-1], s[1:]), 1.0)
    prior = count / count.sum()
    top10 = lambda S, t: (S > S[t]).sum() < 10           # target ranks in the top 10
    pop_hit = np.mean([top10(count, s[-1]) for s in seqs])
    mc_hit = np.mean([top10((T[s[-2]] + prior) / (T[s[-2]].sum() + 1), s[-1]) for s in seqs])
    return {"popularity": round(float(pop_hit), 4), "Markov": round(float(mc_hit), 4)}

for p in (0.0, 0.2, 0.5, 0.8):
    print(f"p_follow={p:.1f}: recall@10 {experiment(p)}")
```

```output
p_follow=0.0: recall@10 {'popularity': 0.256, 'Markov': 0.233}
p_follow=0.2: recall@10 {'popularity': 0.2345, 'Markov': 0.3825}
p_follow=0.5: recall@10 {'popularity': 0.1665, 'Markov': 0.6195}
p_follow=0.8: recall@10 {'popularity': 0.125, 'Markov': 0.8435}
```

At `p_follow=0` there is no structure, and the chain loses to popularity (0.233 against 0.256):
90,000 cells estimating noise. At 0.2 it is already 1.6x popularity; at 0.8 it is near the ceiling.
Add a skip-one rule or shorten the sequences to watch the transition table starve for data — the
same starvation that makes cold-start items (ch. 9) hard.

## Exercise

The SASRec paper trained with binary cross-entropy: per position, the true next item against one
uniform random negative. Train `SASRec()` that way and compare with the full-softmax model on
full-ranking recall/NDCG and sampled HR@10. How much is the loss — and would sampling have told you?

<details>
<summary>Solution</summary>

```python
def bce_one_negative(model, h, y):
    neg = torch.randint(0, n_items, y.shape)              # SASRec's original: 1 uniform negative
    E = model.emb.weight
    return -(F.logsigmoid((h * E[y]).sum(-1)) + F.logsigmoid(-(h * E[neg]).sum(-1))).mean()

sas_bce = train_seq_model(SASRec(), loss_fn=bce_one_negative)
rk_bce = ranks_of_target(scorer(sas_bce), test)
print(f"{'SASRec loss':>22} {'recall@10':>10} {'ndcg@10':>9} {'HR@10, 100 sampled':>19}")
for label, model_, rk in (("full softmax (CE)", sas, results["SASRec"]),
                          ("BCE, 1 negative", sas_bce, rk_bce)):
    r, n = metrics(rk)
    hr_s = float((sampled_ranks(scorer(model_), test) <= 10).mean())
    print(f"{label:>22} {r:>10.4f} {n:>9.4f} {hr_s:>19.4f}")
```

```output
           SASRec loss  recall@10   ndcg@10  HR@10, 100 sampled
     full softmax (CE)     0.5713    0.3500              0.8093
       BCE, 1 negative     0.4960    0.2769              0.7943
```

One-negative BCE costs 0.075 recall and 0.073 NDCG under full ranking — a 21% relative NDCG loss —
but only 0.015 sampled HR@10. One uniform negative is almost always easy, so the model learns to
beat random items — what the sampled metric checks. That is Klenitskiy and Vasilev (2023) in
miniature: the loss moves SASRec more than the gap between architectures, and sampling hides it. Use
the full softmax while the catalogue allows; past that, many negatives with logQ (ch. 7).
