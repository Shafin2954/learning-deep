## From Linear to Neural Q-Functions

Linear function approximation with tile coding works for low-dimensional continuous spaces.
For high-dimensional inputs — raw pixels, text, joint positions — you need neural networks.
Replace $w^T\phi(s)$ with a neural network $Q(s,a;\theta)$ trained by stochastic gradient
descent on a loss derived from the TD target.

The challenge: naive neural Q-learning diverges. Three instabilities work against you:

1. **Temporal correlations**: consecutive transitions are highly correlated, violating the
   i.i.d. assumption SGD needs.
2. **Moving targets**: the TD target $r + \gamma\max_{a'}Q(s',a';\theta)$ uses $\theta$ itself,
   so every update shifts the target, causing oscillations.
3. **Distribution shift**: as the policy improves, the state distribution changes, destabilising
   training.

DQN introduced two fixes for (1) and (2) that made deep RL practical.

## Experience Replay

Store each transition $(s, a, r, s', \text{done})$ in a **replay buffer**. At each training
step, sample a random **minibatch** of transitions from the buffer.

Benefits:
- Breaks temporal correlations (random samples are approximately i.i.d.).
- Each transition can be reused multiple times (data efficiency).
- Decouples data collection from learning rate.

```python
import numpy as np
from collections import deque
np.random.seed(0)

class ReplayBuffer:
    def __init__(self, capacity=10_000):
        self.buf = deque(maxlen=capacity)

    def push(self, s, a, r, ns, done):
        self.buf.append((s, a, r, ns, done))

    def sample(self, batch_size):
        idx = np.random.choice(len(self.buf), batch_size, replace=False)
        batch = [self.buf[i] for i in idx]
        s, a, r, ns, d = zip(*batch)
        return (np.array(s, dtype=np.float32),
                np.array(a), np.array(r, dtype=np.float32),
                np.array(ns, dtype=np.float32),
                np.array(d, dtype=np.float32))

    def __len__(self):
        return len(self.buf)

buf = ReplayBuffer(1000)
for _ in range(200):
    buf.push(np.random.randn(4), np.random.randint(2),
             np.random.randn(), np.random.randn(4), False)

s, a, r, ns, d = buf.sample(32)
print(f"Sampled batch shapes: s={s.shape}, a={a.shape}, r={r.shape}")
```

```output
Sampled batch shapes: s=(32, 4), a=(32,), r=(32,)
```

## Target Network

The second DQN fix: maintain a **target network** $Q(s,a;\theta^-)$ with parameters $\theta^-$
that are a frozen copy of the online network. The loss becomes:

$$
\mathcal{L}(\theta) = \left(r + \gamma \max_{a'} Q(s',a';\theta^-) - Q(s,a;\theta)\right)^2
$$

Every $C$ steps, copy $\theta \rightarrow \theta^-$. This stabilises the target for $C$ steps,
preventing the moving-target oscillation.

## DQN on CartPole

No `gymnasium` is needed here: the block below re-implements CartPole's physics in numpy behind the
same `reset`/`step` interface, so the training loop is exactly what you would write for `gym.make("CartPole-v1")`.
The network is small and the run is capped at 120 episodes so it finishes in under a minute on a CPU.

```python
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from collections import deque
torch.set_num_threads(1)
torch.manual_seed(0)

class CartPole:
    """CartPole-v1 dynamics (Euler integration, tau=0.02) in plain numpy, gymnasium-style API."""
    obs_dim, n_actions = 4, 2
    def __init__(self, seed=0):
        self.rng = np.random.default_rng(seed)
    def reset(self, seed=None):
        if seed is not None:
            self.rng = np.random.default_rng(seed)
        self.s = self.rng.uniform(-0.05, 0.05, 4)
        self.t = 0
        return self.s.copy(), {}
    def sample_action(self):
        return int(self.rng.integers(self.n_actions))
    def step(self, a):
        x, xd, th, thd = self.s
        f = 10.0 if a == 1 else -10.0
        c, sn = np.cos(th), np.sin(th)
        tmp = (f + 0.05 * thd**2 * sn) / 1.1
        thacc = (9.8 * sn - c * tmp) / (0.5 * (4 / 3 - 0.1 * c**2 / 1.1))
        xacc = tmp - 0.05 * thacc * c / 1.1
        self.s = np.array([x + 0.02 * xd, xd + 0.02 * xacc, th + 0.02 * thd, thd + 0.02 * thacc])
        self.t += 1
        terminated = bool(abs(self.s[0]) > 2.4 or abs(self.s[2]) > 0.2095)
        return self.s.copy(), 1.0, terminated, self.t >= 500, {}

class QNet(nn.Module):
    def __init__(self, obs_dim, n_actions):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, 64), nn.ReLU(),
            nn.Linear(64, 64),      nn.ReLU(),
            nn.Linear(64, n_actions)
        )
    def forward(self, x):
        return self.net(x)

env = CartPole(seed=0)
obs_dim, n_actions = env.obs_dim, env.n_actions

online_net = QNet(obs_dim, n_actions)
target_net = QNet(obs_dim, n_actions)
target_net.load_state_dict(online_net.state_dict())
target_net.eval()

optimizer = optim.Adam(online_net.parameters(), lr=1e-3)
buffer = ReplayBuffer(10_000)

gamma = 0.99
epsilon = 1.0
eps_min, eps_decay = 0.05, 0.97
batch_size = 64
target_update = 10
episode_lengths = []

for ep in range(120):
    obs, _ = env.reset()
    total_steps = 0
    done = False
    while not done:
        if np.random.random() < epsilon:
            action = env.sample_action()
        else:
            with torch.no_grad():
                q = online_net(torch.FloatTensor(obs))
            action = q.argmax().item()

        next_obs, reward, terminated, truncated, _ = env.step(action)
        done = terminated or truncated
        buffer.push(obs, action, reward, next_obs, float(terminated))
        obs = next_obs
        total_steps += 1

        if len(buffer) >= batch_size:
            s, a, r, ns, d = buffer.sample(batch_size)
            s_t  = torch.FloatTensor(s)
            a_t  = torch.LongTensor(a)
            r_t  = torch.FloatTensor(r)
            ns_t = torch.FloatTensor(ns)
            d_t  = torch.FloatTensor(d)

            with torch.no_grad():
                target_q = r_t + gamma * target_net(ns_t).max(1).values * (1 - d_t)
            current_q = online_net(s_t).gather(1, a_t.unsqueeze(1)).squeeze()
            loss = nn.MSELoss()(current_q, target_q)

            optimizer.zero_grad()
            loss.backward()
            nn.utils.clip_grad_norm_(online_net.parameters(), 10)
            optimizer.step()

    epsilon = max(eps_min, epsilon * eps_decay)
    episode_lengths.append(total_steps)

    if ep % 10 == 0 and ep > 0:
        target_net.load_state_dict(online_net.state_dict())

    if ep % 20 == 0:
        print(f"Episode {ep:3d}: steps={total_steps:3d}  eps={epsilon:.3f}  "
              f"avg(last 20)={np.mean(episode_lengths[-20:]):.0f}")

```

```output
Episode   0: steps= 10  eps=0.970  avg(last 20)=10
Episode  20: steps= 15  eps=0.527  avg(last 20)=23
Episode  40: steps= 39  eps=0.287  avg(last 20)=29
Episode  60: steps=248  eps=0.156  avg(last 20)=193
Episode  80: steps=254  eps=0.085  avg(last 20)=280
```

The 20-episode average climbs from about 10 steps to roughly 190 by episode 60 and to
about 260 by the end of the 120 episodes (the cap is 500, so this is learning, not yet mastery;
DQN returns are noisy and can dip after peaking). The $\varepsilon$ schedule anneals
exploration from 1.0 to 0.05 as the policy improves. Note that `done` is stored as `terminated`
only: a time-limit truncation is not a real terminal state, so the target should still bootstrap.

## Double DQN

Standard DQN overestimates Q-values: $\max_{a'}Q(s',a';\theta^-)$ both selects and evaluates
the best action with the same (noisy) network, leading to upward bias.

**Double DQN** decouples selection and evaluation:

$$
\text{target} = r + \gamma Q\bigl(s', \arg\max_{a'} Q(s',a';\theta);\, \theta^-\bigr)
$$

Use the online network to pick the action, the target network to evaluate it. One code change:

```python
# Compare the standard and Double DQN targets on one sampled minibatch
# (uses online_net, target_net, buffer, gamma from the DQN block above)
s, a, r, ns, d = buffer.sample(256)
r_t, ns_t, d_t = torch.FloatTensor(r), torch.FloatTensor(ns), torch.FloatTensor(d)

with torch.no_grad():
    std_target = r_t + gamma * target_net(ns_t).max(1).values * (1 - d_t)
    best_actions = online_net(ns_t).argmax(1, keepdim=True)   # online net SELECTS
    dbl_target = r_t + gamma * target_net(ns_t).gather(1, best_actions).squeeze(1) * (1 - d_t)  # target net EVALUATES

print(f"mean standard target: {std_target.mean():.3f}")
print(f"mean double target:   {dbl_target.mean():.3f}")
print(f"double <= standard on every sample: {bool((dbl_target <= std_target + 1e-6).all())}")
```

```output
mean standard target: 11.155
mean double target:   11.126
double <= standard on every sample: True
```

The double target can never exceed the standard one, because the action chosen by the online network
is evaluated by the target network, and the target network's own maximum is by definition at least as large.
Whenever the two networks disagree on the best action, Double DQN gives the smaller (less optimistic) value.

### Dueling DQN

Dueling DQN separates the network into two streams:
- **Value stream** $V(s;\theta)$: the value of being in state $s$.
- **Advantage stream** $A(s,a;\theta)$: how much better action $a$ is than average.

$$
Q(s,a) = V(s) + \left(A(s,a) - \frac{1}{|A|}\sum_{a'} A(s,a')\right)
$$

The mean subtraction prevents the advantage from growing without bound. Dueling DQN learns
$V(s)$ from all transitions (not just the chosen action), which stabilises training.

## Try it Yourself

```python
# Run the DQN above but replace QNet with a Dueling architecture
# (reuses CartPole, ReplayBuffer and the imports from earlier on this page)
torch.manual_seed(1)

class DuelingQNet(nn.Module):
    def __init__(self, obs_dim, n_actions):
        super().__init__()
        self.shared = nn.Sequential(nn.Linear(obs_dim, 64), nn.ReLU())
        self.value  = nn.Sequential(nn.Linear(64, 32), nn.ReLU(), nn.Linear(32, 1))
        self.adv    = nn.Sequential(nn.Linear(64, 32), nn.ReLU(), nn.Linear(32, n_actions))

    def forward(self, x):
        h = self.shared(x)
        v = self.value(h)
        a = self.adv(h)
        return v + a - a.mean(dim=1, keepdim=True)

env = CartPole(seed=0)
net = DuelingQNet(env.obs_dim, env.n_actions)
tgt = DuelingQNet(env.obs_dim, env.n_actions)
tgt.load_state_dict(net.state_dict())

optimizer = optim.Adam(net.parameters(), lr=1e-3)
buf = ReplayBuffer(10_000)
eps = 1.0
lengths = []

for ep in range(120):
    obs, _ = env.reset()
    steps, done = 0, False
    while not done:
        if np.random.random() < eps:
            a = env.sample_action()
        else:
            with torch.no_grad():
                a = net(torch.FloatTensor(obs).unsqueeze(0)).argmax().item()
        no, r, term, trunc, _ = env.step(a)
        done = term or trunc
        buf.push(obs, a, r, no, float(term))
        obs, steps = no, steps + 1
        if len(buf) >= 64:
            s_b,a_b,r_b,ns_b,d_b = buf.sample(64)
            with torch.no_grad():
                tq = torch.FloatTensor(r_b) + 0.99 * tgt(torch.FloatTensor(ns_b)).max(1).values * (1-torch.FloatTensor(d_b))
            cq = net(torch.FloatTensor(s_b)).gather(1, torch.LongTensor(a_b).unsqueeze(1)).squeeze(1)
            loss = nn.MSELoss()(cq, tq)
            optimizer.zero_grad(); loss.backward(); optimizer.step()
    eps = max(0.05, eps * 0.97)
    lengths.append(steps)
    if ep % 10 == 0: tgt.load_state_dict(net.state_dict())

print(f"Standard DQN avg steps (last 20): {np.mean(episode_lengths[-20:]):.0f}")
print(f"Dueling DQN avg steps (last 20):  {np.mean(lengths[-20:]):.0f}")
```

```output
Standard DQN avg steps (last 20): 263
Dueling DQN avg steps (last 20):  168
```

On this seed and budget the dueling network is behind the standard one. A single 120-episode run
is within DQN's run-to-run noise, so don't read this as a verdict; the dueling benefit reported
in the literature shows up on tasks with many actions, where learning $V(s)$ from every
transition matters most.

## Exercise

Implement **Prioritised Experience Replay**: instead of sampling uniformly, sample transition
$i$ with probability $P(i) \propto (|\delta_i| + \epsilon_{prio})^\alpha$ where $\delta_i$ is
the TD error. Use importance sampling weights $w_i = (N \cdot P(i))^{-\beta}$ to correct for
the sampling bias. Verify that your buffer samples high-priority transitions more often and that
the weights down-weight them. (Plugging it into the DQN loop and comparing episodes-to-solve
is a longer experiment; try it after this.)

```python
# PER requires a sum-tree data structure for O(log N) sampling.
# A simplified numpy version (O(N) sampling but correct):
import numpy as np

class PrioritisedBuffer:
    def __init__(self, capacity=10_000, alpha=0.6):
        self.capacity = capacity
        self.buf = []
        self.priorities = np.zeros(capacity, dtype=np.float32)
        self.pos = 0
        self.alpha = alpha

    def push(self, *transition):
        max_p = self.priorities.max() if self.buf else 1.0
        if len(self.buf) < self.capacity:
            self.buf.append(transition)
        else:
            self.buf[self.pos] = transition
        self.priorities[self.pos] = max_p
        self.pos = (self.pos + 1) % self.capacity

    def sample(self, batch_size, beta=0.4):
        n = len(self.buf)
        p = self.priorities[:n] ** self.alpha
        p /= p.sum()
        idx = np.random.choice(n, batch_size, replace=False, p=p)
        weights = (n * p[idx]) ** (-beta)
        weights /= weights.max()
        batch = [self.buf[i] for i in idx]
        s,a,r,ns,d = zip(*batch)
        return (np.array(s,dtype=np.float32), np.array(a), np.array(r,dtype=np.float32),
                np.array(ns,dtype=np.float32), np.array(d,dtype=np.float32),
                idx, weights.astype(np.float32))

    def update_priorities(self, idx, td_errors):
        self.priorities[idx] = np.abs(td_errors) + 1e-6

# Sanity check: give transition 0 a 50x larger TD error than the rest, then sample many times.
pb = PrioritisedBuffer(capacity=100, alpha=0.6)
for i in range(100):
    pb.push(np.zeros(4), 0, 0.0, np.zeros(4), False)
pb.update_priorities(np.arange(100), np.concatenate([[50.0], np.ones(99)]))

np.random.seed(0)
hits = np.zeros(100)
w_first, w_other = [], []
for _ in range(2000):
    *_, idx, weights = pb.sample(10, beta=0.4)
    hits[idx] += 1
    for i, wt in zip(idx, weights):
        (w_first if i == 0 else w_other).append(wt)

print(f"transition 0 sampled in {hits[0] / 2000:.0%} of batches; a typical other one in {hits[1:].mean() / 2000:.0%}")
print(f"IS weight of transition 0: {np.mean(w_first):.2f}   typical other: {np.mean(w_other):.2f}")
```

```output
transition 0 sampled in 67% of batches; a typical other one in 9%
IS weight of transition 0: 0.39   typical other: 1.00
```

The key insight: high-TD-error transitions are more informative, so sample them more often, and
the IS weights (smaller for the over-sampled transitions) keep the update from being biased towards
them. In practice, PER is most beneficial on sparse-reward tasks where most transitions have
near-zero reward. The sum-tree used in real implementations makes sampling $O(\log N)$ instead
of the $O(N)$ here.
