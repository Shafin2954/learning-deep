## Learning Without a Model

Dynamic programming required the transition dynamics $p(s',r|s,a)$. Monte Carlo methods need
nothing of the sort — they learn purely from experience: complete episodes of interaction.
You run the agent, observe the sequence of states, actions, and rewards, and then use the
actual returns to update value estimates.

The trade-off: MC can only update after a complete episode, so it doesn't work for continuing
tasks. But it can handle any environment you can simulate, even one with complex stochastic
dynamics you'd never be able to write down as a model.

## First-Visit MC Prediction

You want to estimate $v_\pi(s) = \mathbb{E}_\pi[G_t \mid S_t = s]$. The return from time $t$
is $G_t = R_{t+1} + \gamma R_{t+2} + \gamma^2 R_{t+3} + \cdots$ over the rest of the episode.

**First-visit MC**: on each episode, for each state $s$, find the *first* time it appears.
Compute $G_t$ from that point. Average over all episodes.

$$
V(s) \leftarrow V(s) + \alpha\bigl(G_t - V(s)\bigr)
$$

Using $\alpha = 1/N(s)$ gives the sample mean. Using a fixed small $\alpha$ tracks
non-stationary problems.

```python
import random
import numpy as np

class Blackjack:
    """Blackjack with Sutton & Barto rules (infinite deck), same interface as gymnasium's Blackjack-v1(sab=True).
    Observation: (player_sum, dealer_showing_card, usable_ace). Actions: 0=stick, 1=hit."""
    def __init__(self, seed=0):
        self.rnd = random.Random(seed)
    def _draw(self):
        return min(self.rnd.randint(1, 13), 10)
    @staticmethod
    def _total(hand):
        t = sum(hand)
        return t + 10 if 1 in hand and t + 10 <= 21 else t
    def _obs(self):
        return (self._total(self.player), self.dealer[0], 1 in self.player and sum(self.player) + 10 <= 21)
    def sample_action(self):
        return self.rnd.randint(0, 1)
    def reset(self):
        self.player = [self._draw(), self._draw()]
        self.dealer = [self._draw(), self._draw()]
        return self._obs(), {}
    def step(self, action):
        if action == 1:  # hit
            self.player.append(self._draw())
            if self._total(self.player) > 21:
                return self._obs(), -1.0, True, False, {}
            return self._obs(), 0.0, False, False, {}
        while self._total(self.dealer) < 17:  # dealer plays out
            self.dealer.append(self._draw())
        p, d = self._total(self.player), self._total(self.dealer)
        if d > 21:
            d = 0  # dealer busts: any standing player wins
        reward = float(p > d) - float(p < d)
        natural = lambda h: sorted(h) == [1, 10]
        if natural(self.player):
            reward = 0.0 if natural(self.dealer) else 1.0
        return self._obs(), reward, True, False, {}

env = Blackjack(seed=0)

def generate_episode(env, policy):
    """Returns list of (state, action, reward)."""
    episode = []
    state, _ = env.reset()
    done = False
    while not done:
        action = policy(state)
        next_state, reward, terminated, truncated, _ = env.step(action)
        episode.append((state, action, reward))
        state = next_state
        done = terminated or truncated
    return episode

def simple_policy(state):
    """Stick (0) if sum >= 20, else hit (1)."""
    player_sum, _, _ = state
    return 0 if player_sum >= 20 else 1

# First-visit MC prediction
returns_sum = {}
returns_count = {}
V = {}

n_episodes = 100_000
for ep in range(n_episodes):
    episode = generate_episode(env, simple_policy)
    G = 0
    visited = set()
    for t in reversed(range(len(episode))):
        state, action, reward = episode[t]
        G = reward + G  # gamma=1 for Blackjack
        if state not in visited:
            visited.add(state)
            returns_sum[state] = returns_sum.get(state, 0) + G
            returns_count[state] = returns_count.get(state, 0) + 1
            V[state] = returns_sum[state] / returns_count[state]

# Show value for usable-ace states with dealer showing 2
print("V(player_sum, dealer=2, usable_ace=True):")
for ps in range(12, 22):
    key = (ps, 2, True)
    val = V.get(key, float('nan'))
    print(f"  player={ps}: {val:.3f}")
```

```output
V(player_sum, dealer=2, usable_ace=True):
  player=12: -0.132
  player=13: -0.258
  player=14: -0.173
  player=15: -0.200
  player=16: -0.306
  player=17: -0.333
  player=18: -0.330
  player=19: -0.300
  player=20: 0.608
  player=21: 0.983
```

This policy hits on everything below 20, so states 12–19 have negative values (hitting a
soft hand often turns it into a worse hard hand or a bust), while the states where it sticks
are strongly positive: $V(20)\approx 0.61$ and $V(21)\approx 0.98$ against a dealer showing 2. The
sharp jump at 20 is the signature of the stick rule, not of a smooth value function. Estimates
for rarely visited states (such as the soft 14) are noisy.

## MC Control: $\varepsilon$-Soft Policy

To find the optimal policy without a model, you need to explore. The **$\varepsilon$-soft** approach
makes every action available with at least probability $\varepsilon/|A|$, but assigns most probability to
the current best action:

$$
\pi(a|s) = \begin{cases}
1 - \varepsilon + \varepsilon/|A| & \text{if } a = \arg\max_{a'} Q(s,a') \\
\varepsilon/|A| & \text{otherwise}
\end{cases}
$$

```python
from collections import defaultdict

env = Blackjack(seed=1)

Q = defaultdict(lambda: np.zeros(2))  # 2 actions: stick=0, hit=1
N = defaultdict(lambda: np.zeros(2))
epsilon = 0.1

def epsilon_greedy(state, eps=epsilon):
    if env.rnd.random() < eps:
        return env.sample_action()
    return int(np.argmax(Q[state]))

n_episodes = 300_000
for ep in range(n_episodes):
    episode = []
    state, _ = env.reset()
    done = False
    while not done:
        action = epsilon_greedy(state)
        next_state, reward, terminated, truncated, _ = env.step(action)
        episode.append((state, action, reward))
        state = next_state
        done = terminated or truncated

    G = 0
    visited = set()
    for t in reversed(range(len(episode))):
        state, action, reward = episode[t]
        G = reward + G
        sa = (state, action)
        if sa not in visited:
            visited.add(sa)
            N[state][action] += 1
            Q[state][action] += (G - Q[state][action]) / N[state][action]

# Show optimal action for no-usable-ace states
print("Optimal action (0=stick, 1=hit) — no usable ace, dealer=6:")
for ps in range(12, 22):
    key = (ps, 6, False)
    action = np.argmax(Q[key])
    print(f"  player={ps}: {'hit' if action else 'stick'}")
```

```output
Optimal action (0=stick, 1=hit) — no usable ace, dealer=6:
  player=12: stick
  player=13: stick
  player=14: stick
  player=15: stick
  player=16: stick
  player=17: stick
  player=18: stick
  player=19: stick
  player=20: stick
  player=21: stick
```

With dealer showing 6 (a "bust card"), the learnt policy matches basic strategy: stick from
12 upward, because the dealer is likely to bust.

### Exploring Starts vs $\varepsilon$-Soft

An alternative to $\varepsilon$-soft is **exploring starts**: guarantee every (state, action) pair has
nonzero probability of being the start of an episode. This allows a fully greedy policy but
requires control over the environment's start distribution — impossible in real settings.
$\varepsilon$-soft is the practical default.

## Importance Sampling for Off-Policy MC

To learn about a **target policy** $\pi$ while following a **behavior policy** $b$ (which
must cover $\pi$: $\pi(a|s) > 0 \Rightarrow b(a|s) > 0$), multiply returns by the
importance sampling ratio:

$$
\rho_{t:T-1} = \prod_{k=t}^{T-1} \frac{\pi(A_k|S_k)}{b(A_k|S_k)}
$$

Ordinary IS: $V(s) \leftarrow \frac{\sum \rho G}{\text{count}}$ — unbiased but high variance.
Weighted IS: $V(s) \leftarrow \frac{\sum \rho G}{\sum \rho}$ — biased but much lower variance
in practice, and the bias vanishes asymptotically.

## Try it Yourself

```python
from collections import defaultdict

# Train MC control on Blackjack for 200,000 episodes with epsilon=0.1
# Then evaluate the greedy policy for 10,000 episodes and report win rate
# (uses the Blackjack class defined earlier on this page)

env = Blackjack(seed=2)
Q = defaultdict(lambda: np.zeros(2))
N = defaultdict(lambda: np.zeros(2))
epsilon = 0.1

for ep in range(200_000):
    episode = []
    state, _ = env.reset()
    done = False
    while not done:
        action = int(np.argmax(Q[state])) if env.rnd.random() > epsilon else env.sample_action()
        next_state, reward, terminated, truncated, _ = env.step(action)
        episode.append((state, action, reward))
        state = next_state
        done = terminated or truncated
    G = 0
    visited = set()
    for t in reversed(range(len(episode))):
        s, a, r = episode[t]
        G = r + G
        if (s, a) not in visited:
            visited.add((s, a))
            N[s][a] += 1
            Q[s][a] += (G - Q[s][a]) / N[s][a]

# Evaluate greedy policy
wins = draws = losses = 0
for _ in range(10_000):
    state, _ = env.reset()
    done = False
    while not done:
        action = int(np.argmax(Q[state]))
        state, reward, terminated, truncated, _ = env.step(action)
        done = terminated or truncated
    if reward > 0:
        wins += 1
    elif reward == 0:
        draws += 1
    else:
        losses += 1

print(f"Win rate: {wins/100:.1f}%  Draw: {draws/100:.1f}%  Loss: {losses/100:.1f}%")
```

```output
Win rate: 42.6%  Draw: 8.8%  Loss: 48.7%
```

## Exercise

Implement **every-visit** MC prediction (not first-visit) on the same Blackjack environment.
After 100,000 episodes, compare the value estimates for states (18, 6, False) and (20, 10, False)
between first-visit and every-visit MC. Are they different? Why or why not in the limit?

<summary>Solution</summary>

```python
env = Blackjack(seed=3)

def simple_policy(state):
    return 0 if state[0] >= 20 else 1

# Every-visit and first-visit MC from the same episodes
returns_sum_ev = {}
returns_count_ev = {}
returns_sum_fv = {}
returns_count_fv = {}

for ep in range(100_000):
    episode = []
    state, _ = env.reset()
    done = False
    while not done:
        action = simple_policy(state)
        ns, reward, terminated, truncated, _ = env.step(action)
        episode.append((state, action, reward))
        state = ns
        done = terminated or truncated
    G = 0
    first_return = {}
    for t in reversed(range(len(episode))):
        s, a, r = episode[t]
        G = r + G
        # Every visit: update even if s was seen before
        returns_sum_ev[s] = returns_sum_ev.get(s, 0) + G
        returns_count_ev[s] = returns_count_ev.get(s, 0) + 1
        first_return[s] = G  # going backwards, the last write is the first visit
    for s, G0 in first_return.items():
        returns_sum_fv[s] = returns_sum_fv.get(s, 0) + G0
        returns_count_fv[s] = returns_count_fv.get(s, 0) + 1

for key in [(18, 6, False), (20, 10, False)]:
    ev = returns_sum_ev.get(key, 0) / max(returns_count_ev.get(key, 1), 1)
    fv = returns_sum_fv.get(key, 0) / max(returns_count_fv.get(key, 1), 1)
    print(f"V{key}: every-visit = {ev:.4f}, first-visit = {fv:.4f}")
```

```output
V(18, 6, False): every-visit = -0.6593, first-visit = -0.6593
V(20, 10, False): every-visit = 0.4296, first-visit = 0.4296
```

In the limit, first-visit and every-visit MC both converge to the true $v_\pi(s)$ — the
difference is small-sample behaviour (every-visit is slightly biased but consistent). Every-visit
uses more updates per episode, but they are correlated within an episode. For Blackjack, a state
never repeats within an episode here (the player's sum only grows), so the two methods agree to
the digits shown above.
</details>
