## Exploration you can act on

Most EDA is decoration. The version that pays produces four decisions: which columns to drop,
which need encoding, where the missingness comes from, and what score a trivial model already
gets. Everything else can wait until you have a baseline to beat.

```python
import numpy as np
import pandas as pd

rng = np.random.default_rng(0)
n = 15_000
zip_code = rng.integers(10_000, 99_999, n)                     # high cardinality, mostly noise
plan = rng.choice(["basic", "plus", "pro", "enterprise"], n, p=[0.5, 0.3, 0.15, 0.05])
tenure = rng.integers(0, 72, n)
usage = np.round(np.exp(rng.normal(3.0, 0.8, n)), 1)
region = rng.choice(["north", "south", "east", "west"], n)
constant_col = np.ones(n)

logit = (-1.1 + 0.035 * tenure - 0.9 * np.log1p(usage) / 3
         + 0.8 * (plan == "basic") - 0.5 * (plan == "enterprise"))
churn = (rng.random(n) < 1 / (1 + np.exp(-logit))).astype(int)

# support_score is missing precisely for customers who never contacted support
support_score = np.where(rng.random(n) < 0.3 + 0.25 * churn, np.nan, rng.normal(70, 12, n))

df = pd.DataFrame({"zip_code": zip_code, "plan": plan, "tenure": tenure, "usage": usage,
                   "region": region, "constant_col": constant_col,
                   "support_score": support_score, "churn": churn})
print(f"{len(df):,} rows x {df.shape[1]} columns | churn rate {df['churn'].mean():.3f}")
print(df.head(3).to_string(index=False))
```

```output

    86555  plus      60   14.9   west           1.0      78.998493      1
    67325 basic       9    6.4   east           1.0            NaN      0

```

## A profile that flags decisions, not statistics

```python
def profile(frame, target):
    rows = []
    for col in frame.columns:
        if col == target:
            continue
        s = frame[col]
        nun = int(s.nunique(dropna=True))
        miss = float(s.isna().mean())
        note = ""
        if nun <= 1:
            note = "DROP: constant"
        elif not pd.api.types.is_numeric_dtype(s) and nun > 0.2 * len(s):
            note = "high-cardinality text: hash or target-encode (ch. 5)"
        elif pd.api.types.is_numeric_dtype(s) and nun > 0.5 * len(s) and s.min() >= 0 \
                and float(s.std()) > 10_000:
            note = "id-like numeric: treat as categorical or drop (ch. 1)"
        elif miss > 0.05:
            note = f"missing {miss:.0%}: add an indicator before imputing"
        rows.append({"column": col, "dtype": str(s.dtype), "unique": nun,
                     "missing": round(miss, 3), "action": note or "ok"})
    return pd.DataFrame(rows)

print(profile(df, "churn").to_string(index=False))
```

### Missingness is a feature until proven otherwise

```python
miss = df["support_score"].isna()
print(f"churn rate when support_score is missing : {df.loc[miss, 'churn'].mean():.3f}")
print(f"churn rate when it is present            : {df.loc[~miss, 'churn'].mean():.3f}")
print(f"missing share overall                    : {miss.mean():.3f}")
```

```output
churn rate when support_score is missing : 0.569
churn rate when it is present            : 0.313
missing share overall                    : 0.408
```

The missingness carries more signal than the column's values will. Impute silently and you throw
it away; add `support_score_isna` and the model keeps it. This is the single most common free
score in tabular competitions.

## The baseline ladder

Climb four rungs before you tune anything. Each rung answers a question the next one depends on.

```python
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import roc_auc_score
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer

y = df["churn"].values
X = df.drop(columns=["churn", "constant_col"])
X = X.assign(support_score_isna=X["support_score"].isna().astype(int))

def cv_auc(make_model, frame, y, k=5, seed=0):
    cv = StratifiedKFold(n_splits=k, shuffle=True, random_state=seed)
    scores = []
    for tr, va in cv.split(frame, y):
        m = make_model().fit(frame.iloc[tr], y[tr])
        scores.append(roc_auc_score(y[va], m.predict_proba(frame.iloc[va])[:, 1]))
    return float(np.mean(scores)), float(np.std(scores))

num_cols = ["tenure", "usage", "support_score", "support_score_isna", "zip_code"]
cat_cols = ["plan", "region"]
pre = ColumnTransformer([
    ("num", make_pipeline(SimpleImputer(strategy="median"), StandardScaler()), num_cols),
    ("cat", OneHotEncoder(handle_unknown="ignore"), cat_cols),
])

class Constant:
    def fit(self, X, y=None): return self
    def predict_proba(self, X): return np.column_stack([np.full(len(X), 0.5)] * 2)

ladder = [
    ("0. constant", lambda: Constant(), X),
    ("1. one feature (tenure)", lambda: make_pipeline(StandardScaler(),
                                                      LogisticRegression()), X[["tenure"]]),
    ("2. logistic on everything", lambda: make_pipeline(pre, LogisticRegression(max_iter=1_000)), X),
    ("3. gradient boosting", lambda: make_pipeline(pre, HistGradientBoostingClassifier(
        random_state=0)), X),
]
results = {}
for name, make, frame in ladder:
    mean, sd = cv_auc(make, frame, y)
    results[name] = mean
    print(f"{name:28} AUC {mean:.4f} +- {sd:.4f}")
```

```output
0. constant                  AUC 0.5000 +- 0.0000
1. one feature (tenure)      AUC 0.6865 +- 0.0058
2. logistic on everything    AUC 0.7582 +- 0.0054
3. gradient boosting         AUC 0.7510 +- 0.0054
```

Read the *gaps*, not the scores:

- **rung 0 to 1** — is there any signal at all in the most obvious column?
- **rung 1 to 2** — how much do the other features add on top of it?
- **rung 2 to 3** — how much is non-linear or interaction-driven? A small gap means the problem is
  mostly additive, and your effort belongs in features rather than in a bigger model.

```python
print(f"signal exists      (0 -> 1): {results['1. one feature (tenure)'] - 0.5:+.4f}")
print(f"other features add (1 -> 2): "
      f"{results['2. logistic on everything'] - results['1. one feature (tenure)']:+.4f}")
print(f"non-linearity adds (2 -> 3): "
      f"{results['3. gradient boosting'] - results['2. logistic on everything']:+.4f}")
```

```output
signal exists      (0 -> 1): +0.1865
other features add (1 -> 2): +0.0717
non-linearity adds (2 -> 3): -0.0072
```

## Pick an iteration size from the learning curve

You will run hundreds of experiments. Run them on the smallest subsample whose ranking still
matches the full data.

```python
sizes = [1_000, 3_000, 7_000, 15_000]
curve = []
for size in sizes:
    idx = np.random.default_rng(1).permutation(len(y))[:size]
    sub, ysub = X.iloc[idx], y[idx]
    a_lin, _ = cv_auc(lambda: make_pipeline(pre, LogisticRegression(max_iter=1_000)), sub, ysub)
    a_gbm, _ = cv_auc(lambda: make_pipeline(pre, HistGradientBoostingClassifier(random_state=0)),
                      sub, ysub)
    curve.append((size, a_lin, a_gbm))
    print(f"n={size:>6,} | logistic {a_lin:.4f} | gbm {a_gbm:.4f} | "
          f"gap {a_gbm - a_lin:+.4f} | gbm wins: {a_gbm > a_lin}")
```

```output
n= 1,000 | logistic 0.7409 | gbm 0.6722 | gap -0.0687 | gbm wins: False
n= 3,000 | logistic 0.7546 | gbm 0.7088 | gap -0.0457 | gbm wins: False
n= 7,000 | logistic 0.7532 | gbm 0.7280 | gap -0.0252 | gbm wins: False
n=15,000 | logistic 0.7577 | gbm 0.7522 | gap -0.0055 | gbm wins: False
```

Here the ordering is stable from 1,000 rows up — logistic regression wins at every size — so a
2,000-row subsample is enough for iteration. But read the gap column too: it closes from -0.069 to
-0.006 as the data grows, which says the boosted model is data-starved rather than wrong. A
competition with 150,000 rows instead of 15,000 would very likely flip this table, and that is
worth knowing before you write off a model family.

The rule: if the *ordering* is stable from some size upward, iterate at that size and re-run only
the finalists on everything. If the ordering flips as the data grows, you are reading noise and
the subsample is too small.

## Exercise

Write the script you run in the first hour of a competition: profile the data, build the baseline
ladder, size the iteration subsample from a learning curve, and end with a short memo that states
what to do next and why. The memo must be generated from the numbers, not written by hand.

```python
def first_hour(frame, target, num_cols, cat_cols, sizes=(2_000, 6_000)):
    y = frame[target].values
    X = frame.drop(columns=[target])
    prof = profile(frame, target)
    drops = prof.loc[prof["action"].str.startswith("DROP"), "column"].tolist()
    X = X.drop(columns=drops)
    added = []
    for col in num_cols:
        if col in X and X[col].isna().mean() > 0.02:
            X[f"{col}_isna"] = X[col].isna().astype(int)
            added.append(f"{col}_isna")
    use_num = [c for c in num_cols + added if c in X]
    pre_local = ColumnTransformer([
        ("num", make_pipeline(SimpleImputer(strategy="median"), StandardScaler()), use_num),
        ("cat", OneHotEncoder(handle_unknown="ignore"), [c for c in cat_cols if c in X]),
    ])
    lin = lambda: make_pipeline(pre_local, LogisticRegression(max_iter=1_000))
    gbm = lambda: make_pipeline(pre_local, HistGradientBoostingClassifier(random_state=0))
    best_single = max(((c, abs(float(np.corrcoef(X[c].fillna(X[c].median()), y)[0, 1])))
                       for c in use_num), key=lambda kv: kv[1])[0]
    one = lambda: make_pipeline(SimpleImputer(strategy="median"), StandardScaler(),
                                LogisticRegression())
    rungs = {
        "one feature": cv_auc(one, X[[best_single]], y)[0],
        "linear": cv_auc(lin, X, y)[0],
        "gbm": cv_auc(gbm, X, y)[0],
    }
    stable_at = None
    for size in sizes:
        idx = np.random.default_rng(1).permutation(len(y))[:size]
        a_lin = cv_auc(lin, X.iloc[idx], y[idx])[0]
        a_gbm = cv_auc(gbm, X.iloc[idx], y[idx])[0]
        if (a_gbm > a_lin) == (rungs["gbm"] > rungs["linear"]) and stable_at is None:
            stable_at = size

    print(f"dropped        : {drops or 'nothing'}")
    print(f"added          : {added or 'nothing'}")
    print(f"strongest single feature: {best_single}")
    for k, v in rungs.items():
        print(f"  {k:12} AUC {v:.4f}")
    nonlin = rungs["gbm"] - rungs["linear"]
    extra = rungs["linear"] - rungs["one feature"]
    print(f"\nMEMO")
    print(f"- signal is real: one feature already reaches {rungs['one feature']:.4f}")
    print(f"- the rest of the table adds {extra:+.4f}; "
          f"{'worth engineering more features' if extra > 0.02 else 'features are thin'}")
    print(f"- non-linearity adds {nonlin:+.4f}: "
          f"{'use a GBDT and spend effort on interactions' if nonlin > 0.01 else 'the problem is additive; a linear model plus good features may match a GBDT'}")
    print(f"- iterate at n={stable_at or max(sizes)}, confirm finalists on all {len(y):,} rows")
    print(f"- next: target-encode high-cardinality columns (ch. 5), "
          f"then adversarial validation (ch. 10)")

first_hour(df, "churn", num_cols=["tenure", "usage", "support_score", "zip_code"],
           cat_cols=["plan", "region"])
```

The memo is generated, which is the entire discipline: every claim in it traces to a number
printed above it, so it cannot drift from the data as the week goes on. Rerun the script after any
change to the feature set and the memo updates itself.

Three judgements are encoded in the thresholds and you should tune them to your own experience.
The 0.02 gap that decides "worth engineering more features" is a competition-scale judgement — on
a leaderboard where the top hundred teams sit inside 0.002 AUC, that bar is far too high and you
would set it an order of magnitude lower. The 0.01 non-linearity bar decides whether you reach for
a GBDT at all, and it is the one that saves the most time: when the gap is genuinely near zero, a
regularised linear model on well-built features trains in seconds, blends beautifully with trees
(ch. 7), and never overfits a small competition.

The missingness handling deserves the last word. The script adds an indicator for any column above
2% missing *before* imputing, which is cheap insurance against exactly the failure this chapter
opened with: a column whose absence predicts the target better than its values do. Note that it
adds the indicator unconditionally rather than testing whether the missingness correlates with the
target — testing it on the full training set is a leak (eval-tuning ch. 5), and the indicator costs
one column.
