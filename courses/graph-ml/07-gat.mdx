## Why attention on graphs?

GCN uses symmetric normalization $1/\sqrt{d_v d_u}$ — a fixed structural weight that ignores
feature content. GraphSAGE uses uniform aggregation over sampled neighbors. Neither lets the
model learn that some neighbors matter more than others for a given node.

Graph Attention Networks (GAT, Velickovic et al., 2018) learn attention weights $\alpha_{vu}$
that determine how much node $v$ attends to each neighbor $u$:

$$
h_v^{(k)} = \sigma\!\left(\sum_{u \in \tilde{\mathcal{N}}(v)} \alpha_{vu} \cdot W h_u^{(k-1)}\right)
$$

The weights $\alpha_{vu}$ are learned from node features, not fixed by structure.

## Computing attention coefficients

Raw attention score between $v$ and $u$:

$$
e_{vu} = \text{LeakyReLU}\!\left(\vec{a}^T \left[Wh_v \;\|\; Wh_u\right]\right)
$$

Normalized via softmax over $v$'s neighborhood:

$$
\alpha_{vu} = \frac{\exp(e_{vu})}{\sum_{k \in \tilde{\mathcal{N}}(v)} \exp(e_{vk})}
$$

where $\vec{a} \in \mathbb{R}^{2d'}$ is a learnable attention vector and $W \in \mathbb{R}^{d' \times d}$
projects features into the attention space.

## Multi-head attention

Run $K$ independent attention heads, then concatenate (hidden layers) or average (output layer):

$$
h_v^{(k)} = \|_{k=1}^{K}\, \sigma\!\left(\sum_{u \in \tilde{\mathcal{N}}(v)} \alpha_{vu}^{(k)} W^{(k)} h_u\right)
$$

Multiple heads let the model capture different types of neighbor relationships simultaneously.

## GAT from scratch

`torch_geometric` is not available here, so the layer is written in plain PyTorch. It adds self-loops (so $\tilde{\mathcal{N}}(v)$ includes $v$), computes one score per edge and head, applies a softmax over each destination node's incoming edges, and aggregates. A flag switches between the original GAT score and the GATv2 score (used later in this chapter).

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class GATLayerScratch(nn.Module):
    def __init__(self, in_ch, out_ch, heads=1, concat=True, dropout=0.6, v2=False):
        super().__init__()
        self.heads, self.out_ch, self.concat, self.v2 = heads, out_ch, concat, v2
        self.W = nn.Linear(in_ch, heads * out_ch, bias=False)
        if v2:   # GATv2: a^T LeakyReLU(W_l h_v + W_r h_u)
            self.W_r = nn.Linear(in_ch, heads * out_ch, bias=False)
            self.a = nn.Parameter(torch.empty(heads, out_ch))
        else:    # GAT: LeakyReLU(a^T [W h_v || W h_u])
            self.a = nn.Parameter(torch.empty(heads, 2 * out_ch))
        nn.init.xavier_uniform_(self.a)
        self.leaky = nn.LeakyReLU(0.2)
        self.dropout = nn.Dropout(dropout)
        self.last_alpha = None

    def forward(self, x, edge_index):
        N = x.size(0)
        loops = torch.arange(N)
        src = torch.cat([edge_index[0], loops])        # add self-loops
        dst = torch.cat([edge_index[1], loops])

        Wx = self.W(x).view(N, self.heads, self.out_ch)            # (N, H, C)
        if self.v2:
            Wr = self.W_r(x).view(N, self.heads, self.out_ch)
            e = (self.leaky(Wx[dst] + Wr[src]) * self.a).sum(-1)   # (E, H)
        else:
            e = torch.cat([Wx[dst], Wx[src]], dim=-1)              # (E, H, 2C)
            e = self.leaky((e * self.a).sum(-1))                   # (E, H)

        # Softmax over each node's incoming edges (subtract per-node max for stability)
        e_max = torch.full((N, self.heads), float('-inf')).scatter_reduce(
            0, dst[:, None].expand(-1, self.heads), e, reduce='amax', include_self=True)
        exp_e = torch.exp(e - e_max[dst])
        denom = torch.zeros(N, self.heads).index_add_(0, dst, exp_e)
        alpha = exp_e / denom[dst]                                 # (E, H)
        self.last_alpha = (src, dst, alpha.detach())
        alpha = self.dropout(alpha)

        out = torch.zeros(N, self.heads, self.out_ch)
        out.index_add_(0, dst, alpha.unsqueeze(-1) * Wx[src])
        return out.reshape(N, -1) if self.concat else out.mean(dim=1)

print("GATLayerScratch ready.")
```

## Node classification on a synthetic citation graph

The data is a small stochastic block model with 5 classes of 60 nodes and 100 noisy bag-of-words features (a Cora stand-in that needs no download), with 10 labeled nodes per class.

```python
import numpy as np
import torch
torch.set_num_threads(1)

def make_graph(n_per=60, k=5, p_in=0.06, p_out=0.015, d=100, seed=2):
    """Stochastic block model + noisy bag-of-words features (a tiny Cora stand-in)."""
    rng = np.random.default_rng(seed)
    y = np.repeat(np.arange(k), n_per)
    n = len(y)
    P = np.where(y[:, None] == y[None, :], p_in, p_out)
    U = np.triu(rng.random((n, n)) < P, 1)
    A = U | U.T
    base = np.full((k, d), 0.03)
    for c in range(k):
        base[c, c * 20:(c + 1) * 20] = 0.15        # class-favored words
    X = (rng.random((n, d)) < base[y]).astype(np.float32)
    X = X / np.maximum(X.sum(1, keepdims=True), 1)
    train = np.zeros(n, bool)
    for c in range(k):
        train[np.nonzero(y == c)[0][:10]] = True
    rest = np.nonzero(~train)[0]
    rng.shuffle(rest)
    val = np.zeros(n, bool); val[rest[:100]] = True
    test = ~train & ~val
    return (torch.tensor(X), torch.tensor(np.stack(np.nonzero(A))), torch.tensor(y),
            torch.tensor(train), torch.tensor(val), torch.tensor(test))

x, edge_index, y, train_mask, val_mask, test_mask = make_graph()
num_nodes, num_classes = x.shape[0], 5


class GAT(nn.Module):
    def __init__(self, hidden=8, heads=8, v2=False, dropout=0.6):
        super().__init__()
        self.l1 = GATLayerScratch(x.shape[1], hidden, heads=heads, concat=True, dropout=dropout, v2=v2)
        self.l2 = GATLayerScratch(hidden * heads, num_classes, heads=1, concat=False, dropout=dropout, v2=v2)
        self.p = dropout

    def forward(self, x, edge_index):
        x = F.dropout(x, self.p, self.training)
        x = F.elu(self.l1(x, edge_index))
        x = F.dropout(x, self.p, self.training)
        return self.l2(x, edge_index)

def fit(model, mask=train_mask, epochs=200, lr=0.005):
    opt = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=5e-4)
    for _ in range(epochs):
        model.train(); opt.zero_grad()
        F.cross_entropy(model(x, edge_index)[mask], y[mask]).backward()
        opt.step()
    model.eval()
    with torch.no_grad():
        pred = model(x, edge_index).argmax(1)
    return (pred[test_mask] == y[test_mask]).float().mean().item()

torch.manual_seed(0)
model = GAT()
print(f"GAT (from scratch) test accuracy: {fit(model):.3f}")
```

```output
GAT (from scratch) test accuracy: 0.680
```

## Attention inspection

The scratch layer stores the attention weights from its last forward pass, so you can look at how a node distributes its attention.

```python
src_all, dst_all, alpha = model.l1.last_alpha
print(f"Attention weight tensor shape: {tuple(alpha.shape)}")
print(f"  (edges incl. self-loops x heads) = ({alpha.shape[0]} edges, {alpha.shape[1]} heads)")

# Find all edges where the target node is the destination
target_node = 0
incoming = (dst_all == target_node)
neighbors = src_all[incoming]
weights_head0 = alpha[incoming, 0]

print(f"\nNode {target_node} (class {y[target_node].item()}): neighbors and attention (head 0):")
for i in weights_head0.argsort(descending=True)[:5]:
    n = neighbors[i].item()
    print(f"  Neighbor {n:3d} (class {y[n].item()})  alpha={weights_head0[i].item():.4f}")
print(f"Uniform attention would give each of the {len(neighbors)} inputs {1/len(neighbors):.4f}")
```

The attention stays close to uniform (0.111 per input for this node) and only weakly favors same-class neighbors: the layer has only seen 50 labels, so it has learned little about which neighbors to trust. Real learned attention on a larger graph is usually sharper.

## GATv2: fixing static attention

The original GAT computes attention as a linear function of the concatenation $[Wh_i, Wh_j]$.
This means the ranking of a node's neighbors doesn't depend on the query node's content
(static attention). **GATv2** fixes this by applying the nonlinearity before the attention
vector:

$$
e_{vu}^{v2} = \vec{a}^T \text{LeakyReLU}\!\left(W [h_v \;\|\; h_u]\right)
$$

The LeakyReLU inside makes attention *dynamic*: different queries can rank the same neighbor set
differently. The scratch layer above already supports it through `v2=True`.

```python
accs = {}
for v2 in [False, True]:
    runs = []
    for seed in range(3):
        torch.manual_seed(seed)
        runs.append(fit(GAT(v2=v2)))
    accs[v2] = sum(runs) / 3
print(f"GAT   mean test accuracy over 3 seeds: {accs[False]:.3f}")
print(f"GATv2 mean test accuracy over 3 seeds: {accs[True]:.3f}")
```

```output
GAT   mean test accuracy over 3 seeds: 0.693
GATv2 mean test accuracy over 3 seeds: 0.678
```

## Exercise

Compare GAT and GCN under **reduced training data**: train with 10 labeled nodes per class
(the default here), then with only 3 per class. Does GAT's attention mechanism help more
with less labeled data? Why or why not?

```python
class DenseGCN(nn.Module):
    def __init__(self, hidden=32):
        super().__init__()
        A = torch.zeros(num_nodes, num_nodes)
        A[edge_index[0], edge_index[1]] = 1.0
        A = A + torch.eye(num_nodes)
        d = A.sum(1).pow(-0.5)
        self.A_hat = d[:, None] * A * d[None, :]
        self.l1, self.l2 = nn.Linear(x.shape[1], hidden), nn.Linear(hidden, num_classes)
    def forward(self, x, edge_index):
        h = F.dropout(F.relu(self.A_hat @ self.l1(x)), 0.5, self.training)
        return self.A_hat @ self.l2(h)

def small_mask(n_per_class):
    mask = torch.zeros(num_nodes, dtype=torch.bool)
    for c in range(num_classes):
        mask[torch.nonzero((y == c) & train_mask).squeeze(1)[:n_per_class]] = True
    return mask

for n in [10, 3]:
    mask = small_mask(n)
    res = {}
    for name, cls in [('GCN', DenseGCN), ('GAT', GAT)]:
        runs = []
        for seed in range(5):
            torch.manual_seed(seed)
            runs.append(fit(cls(), mask=mask))
        res[name] = sum(runs) / 5
    print(f"{n:2d} labels/class: GCN={res['GCN']:.3f}  GAT={res['GAT']:.3f}")
```

```output
10 labels/class: GCN=0.733  GAT=0.693
 3 labels/class: GCN=0.583  GAT=0.565
```

On this small, noisy graph attention does not help: GCN is ahead of GAT at both label budgets (0.733 vs 0.693 with 10 labels per class, 0.583 vs 0.565 with 3), and both lose accuracy as labels shrink. GAT has many more parameters to fit from the same few labels, and in this block model the within-class and cross-class edges look alike in feature space, so there is little for attention to exploit; the fixed degree-based weights of GCN are a stronger prior. Attention pays off when some neighbors are clearly more informative than others, and it can hurt when labels are scarce and neighborhoods are uninformative.
