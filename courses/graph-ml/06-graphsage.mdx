## The inductive problem

GCN is transductive: it requires all nodes and their features to be present during training.
New nodes added after training cannot be embedded without retraining the entire model.

GraphSAGE (Hamilton et al., 2017) is **inductive**: it learns an aggregation *function* rather
than per-node embeddings. Given a new node's features and neighborhood, the learned aggregator
produces its embedding on the fly.

The update rule for node $v$ at layer $k$:

$$
h_{\mathcal{N}(v)}^{(k)} = \text{AGGREGATE}^{(k)}\!\left(\left\{h_u^{(k-1)} : u \in \mathcal{N}(v)\right\}\right)
$$

$$
h_v^{(k)} = \sigma\!\left(W^{(k)} \cdot \left[h_v^{(k-1)} \;\|\; h_{\mathcal{N}(v)}^{(k)}\right]\right)
$$

The key difference from GCN: GraphSAGE concatenates the node's own previous representation
with the neighborhood aggregate, preserving the node's identity.

## Neighbor sampling

Training on the full neighborhood is expensive for high-degree nodes. GraphSAGE samples a
fixed number of neighbors per layer:

- Layer 1: sample $K_1$ neighbors per node (the paper uses 25; the code below uses 10)
- Layer 2: sample $K_2$ neighbors per node (the paper uses 10; the code below uses 5)

This makes memory cost per mini-batch constant: $B \times K_1 \times K_2$ nodes maximum.

No `torch_geometric` here, so everything below is plain PyTorch. The data is a small synthetic citation graph: a stochastic block model with 5 classes of 60 nodes (same-class edge probability 0.06, cross-class 0.015) and 100 noisy bag-of-words features, with 10 labeled nodes per class for training.

```python
import numpy as np
import torch
torch.set_num_threads(1)

def make_graph(n_per=60, k=5, p_in=0.06, p_out=0.015, d=100, seed=2):
    """Stochastic block model + noisy bag-of-words features (a tiny Cora stand-in)."""
    rng = np.random.default_rng(seed)
    y = np.repeat(np.arange(k), n_per)
    n = len(y)
    P = np.where(y[:, None] == y[None, :], p_in, p_out)
    U = np.triu(rng.random((n, n)) < P, 1)
    A = U | U.T
    base = np.full((k, d), 0.03)
    for c in range(k):
        base[c, c * 20:(c + 1) * 20] = 0.15        # class-favored words
    X = (rng.random((n, d)) < base[y]).astype(np.float32)
    X = X / np.maximum(X.sum(1, keepdims=True), 1)
    train = np.zeros(n, bool)
    for c in range(k):
        train[np.nonzero(y == c)[0][:10]] = True
    rest = np.nonzero(~train)[0]
    rng.shuffle(rest)
    val = np.zeros(n, bool); val[rest[:100]] = True
    test = ~train & ~val
    return (torch.tensor(X), torch.tensor(np.stack(np.nonzero(A))), torch.tensor(y),
            torch.tensor(train), torch.tensor(val), torch.tensor(test))

x, edge_index, y, train_mask, val_mask, test_mask = make_graph()
num_nodes, num_classes = x.shape[0], 5

print(f"Nodes: {num_nodes}, directed edges: {edge_index.shape[1]}, train/val/test: "
      f"{int(train_mask.sum())}/{int(val_mask.sum())}/{int(test_mask.sum())}")

def build_neighbors(edge_index, n):
    """neighbors[v] = tensor of u with an edge u -> v."""
    nbrs = [[] for _ in range(n)]
    for u, v in edge_index.t().tolist():
        nbrs[v].append(u)
    return [torch.tensor(l, dtype=torch.long) for l in nbrs]

neighbors = build_neighbors(edge_index, num_nodes)

def sample_subgraph(seeds, fanouts, gen):
    """Layer-wise neighbor sampling. Returns (nodes, edge_index in local ids); seeds come first."""
    nodes = list(seeds.tolist()); index = {v: i for i, v in enumerate(nodes)}
    frontier, edges = seeds.tolist(), []
    for k in fanouts:
        new_frontier = []
        for v in frontier:
            nb = neighbors[v]
            if len(nb) > k:
                nb = nb[torch.randperm(len(nb), generator=gen)[:k]]
            for u in nb.tolist():
                if u not in index:
                    index[u] = len(nodes); nodes.append(u); new_frontier.append(u)
                edges.append((index[u], index[v]))
        frontier = new_frontier
    return torch.tensor(nodes), torch.tensor(edges).t()

gen = torch.Generator().manual_seed(0)
seeds = torch.nonzero(train_mask).squeeze(1)[:25]
nodes, sub_ei = sample_subgraph(seeds, [10, 5], gen)
print(f"Batch nodes (includes neighbors): {len(nodes)}")
print(f"Seed nodes (labeled):             {len(seeds)}")
print(f"Edge index shape:                 {tuple(sub_ei.shape)}")
```

```output
Nodes: 300, directed edges: 2174, train/val/test: 50/100/150
Batch nodes (includes neighbors): 258
Seed nodes (labeled):             25
Edge index shape:                 (2, 652)
```

## Building GraphSAGE in plain PyTorch

`SAGEConv` below implements the update rule directly: aggregate neighbors (mean or element-wise max), concatenate with the node's own vector, apply a linear map.

```python
import torch.nn as nn
import torch.nn.functional as F

class SAGEConv(nn.Module):
    def __init__(self, in_ch, out_ch, aggr='mean'):
        super().__init__()
        self.aggr, self.lin = aggr, nn.Linear(2 * in_ch, out_ch)

    def forward(self, h, edge_index):
        src, dst = edge_index
        agg = torch.zeros_like(h)
        if self.aggr == 'mean':
            agg.index_add_(0, dst, h[src])
            cnt = torch.zeros(h.shape[0], 1).index_add_(0, dst, torch.ones(src.numel(), 1))
            agg = agg / cnt.clamp(min=1)
        else:  # max; nodes without neighbors stay at 0
            agg.scatter_reduce_(0, dst[:, None].expand(-1, h.shape[1]), h[src],
                                reduce='amax', include_self=False)
        return self.lin(torch.cat([h, agg], dim=1))

class GraphSAGE(nn.Module):
    def __init__(self, in_ch, hidden, out_ch, aggr='mean', layers=2):
        super().__init__()
        dims = [in_ch] + [hidden] * (layers - 1) + [out_ch]
        self.convs = nn.ModuleList(SAGEConv(dims[i], dims[i + 1], aggr) for i in range(layers))

    def forward(self, x, edge_index):
        for conv in self.convs[:-1]:
            x = F.dropout(F.relu(conv(x, edge_index)), p=0.5, training=self.training)
        return self.convs[-1](x, edge_index)

torch.manual_seed(0)
gen = torch.Generator().manual_seed(0)
model = GraphSAGE(x.shape[1], 32, num_classes)
optimizer = torch.optim.Adam(model.parameters(), lr=0.005, weight_decay=5e-4)
train_nodes = torch.nonzero(train_mask).squeeze(1)

def train_minibatch(batch_size=25, fanouts=(10, 5)):
    model.train()
    perm = train_nodes[torch.randperm(len(train_nodes), generator=gen)]
    total, nb = 0.0, 0
    for i in range(0, len(perm), batch_size):
        seeds = perm[i:i + batch_size]
        nodes, sub_ei = sample_subgraph(seeds, list(fanouts), gen)
        optimizer.zero_grad()
        out = model(x[nodes], sub_ei)
        # Only compute loss on the seed nodes (the first len(seeds) nodes)
        loss = F.cross_entropy(out[:len(seeds)], y[seeds])
        loss.backward(); optimizer.step()
        total += loss.item(); nb += 1
    return total / nb

for epoch in range(1, 51):
    loss = train_minibatch()

print(f"Final mini-batch training loss: {loss:.4f}")
```

```output
Final mini-batch training loss: 0.2716
```

## Evaluation and comparison with GCN

```python
@torch.no_grad()
def evaluate(model, edge_index=edge_index):
    model.eval()
    # Full-graph inference (no sampling at eval time)
    pred = model(x, edge_index).argmax(dim=1)
    return {split: (pred[mask] == y[mask]).float().mean().item()
            for split, mask in [('train', train_mask), ('val', val_mask), ('test', test_mask)]}

accs = evaluate(model)
print(f"GraphSAGE  Train: {accs['train']:.3f}  "
      f"Val: {accs['val']:.3f}  Test: {accs['test']:.3f}")
```

```output
GraphSAGE  Train: 1.000  Val: 0.730  Test: 0.687
```

## Inductive experiment: embedding unseen nodes

The true test of inductiveness: train on 80% of nodes, then embed the remaining 20% using
only their features and edges, with the model never having seen them (nor their edges) during training.

```python
torch.manual_seed(42)
perm = torch.randperm(num_nodes)
train_nodes_ind = perm[:int(0.8 * num_nodes)]
test_nodes = perm[int(0.8 * num_nodes):]

train_mask_ind = torch.zeros(num_nodes, dtype=torch.bool)
train_mask_ind[train_nodes_ind] = True

# Training graph: only edges with both endpoints in the training set
src, dst = edge_index
train_edge_index = edge_index[:, train_mask_ind[src] & train_mask_ind[dst]]

model_ind = GraphSAGE(x.shape[1], 32, num_classes)
opt_ind = torch.optim.Adam(model_ind.parameters(), lr=0.005)

for epoch in range(100):
    model_ind.train(); opt_ind.zero_grad()
    out = model_ind(x, train_edge_index)
    F.cross_entropy(out[train_mask_ind], y[train_mask_ind]).backward(); opt_ind.step()

# Inference on the full graph: unseen nodes now bring their own edges
model_ind.eval()
with torch.no_grad():
    pred = model_ind(x, edge_index).argmax(1)
    test_acc = (pred[test_nodes] == y[test_nodes]).float().mean()
    train_acc = (pred[train_nodes_ind] == y[train_nodes_ind]).float().mean()

print(f"Inductive: Train acc: {train_acc:.3f},  Test (unseen) acc: {test_acc:.3f}")
```

```output
Inductive: Train acc: 0.983,  Test (unseen) acc: 0.817
```

The gap (0.983 train vs 0.817 test) is expected: test nodes were never seen during training, and neither were their edges. The unseen-node accuracy is far above random (1/5 = 0.2), and notably higher than the transductive test accuracy above, because this run trains on 240 labeled nodes instead of 50. What matters is that the same weights embed nodes that did not exist at training time.

## Aggregator variants

```python
for aggr in ['mean', 'max']:
    torch.manual_seed(0)
    m = GraphSAGE(x.shape[1], 32, num_classes, aggr=aggr)
    o = torch.optim.Adam(m.parameters(), lr=0.005, weight_decay=5e-4)
    for _ in range(100):
        m.train(); o.zero_grad()
        out = m(x, edge_index)
        F.cross_entropy(out[train_mask], y[train_mask]).backward(); o.step()
    print(f"Aggregator={aggr:4s}  test acc={evaluate(m)['test']:.3f}")
```

```output
Aggregator=mean  test acc=0.713
Aggregator=max   test acc=0.567
```

Mean beats max here: with sparse, noisy bag-of-words features, averaging neighbors denoises, while an element-wise max picks up the loudest noise in any neighbor.

## Try it Yourself

```python
# Reuses GraphSAGE, sample_subgraph, evaluate from earlier on this page (same kernel session).
# Train with different neighbor fanouts and compare how many nodes each mini-batch touches.
for fanouts in [(10, 5), (3, 2)]:
    torch.manual_seed(0)
    gen = torch.Generator().manual_seed(0)
    m = GraphSAGE(x.shape[1], 32, num_classes)
    o = torch.optim.Adam(m.parameters(), lr=0.005, weight_decay=5e-4)
    touched = 0
    for _ in range(50):
        m.train()
        perm = train_nodes[torch.randperm(len(train_nodes), generator=gen)]
        for i in range(0, len(perm), 25):
            seeds = perm[i:i + 25]
            nodes, sub_ei = sample_subgraph(seeds, list(fanouts), gen)
            touched += len(nodes)
            o.zero_grad()
            F.cross_entropy(m(x[nodes], sub_ei)[:len(seeds)], y[seeds]).backward(); o.step()
    print(f"fanouts={fanouts}: nodes touched over training = {touched}, test acc = {evaluate(m)['test']:.3f}")
```

```output
fanouts=(10, 5): nodes touched over training = 27027, test acc = 0.687
fanouts=(3, 2): nodes touched over training = 14678, test acc = 0.667
```

Smaller fanouts touch fewer nodes per batch (cheaper) at some cost in gradient quality.


Neither change helps on this graph: the 2-layer, 32-wide model is best (0.713), the third layer drops accuracy to 0.662 (each extra hop mixes in more cross-class neighbors, the start of over-smoothing, on a graph where only about half the edges are within-class), and widening to 128 gives 0.696, within noise of the baseline because the 100-dim input is already small relative to 32 hidden units and only 50 labels are available.
