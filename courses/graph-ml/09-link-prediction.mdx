## The task

Given a graph at time $t$, predict which edges will appear at time $t+1$, or equivalently,
which pairs of nodes should be connected that aren't yet. Applications: friend recommendation,
knowledge graph completion, drug-target interaction prediction.

The standard formulation: learn a scoring function $s: V \times V \rightarrow \mathbb{R}$ such
that $s(u,v) > s(u,w)$ whenever $(u,v)$ is a true edge and $(u,w)$ is not.

## Scoring functions

Given node embeddings $h_u, h_v \in \mathbb{R}^d$:

| Scorer | Formula | Parameters |
|--------|---------|-----------|
| Dot product | $h_u^T h_v$ | 0 |
| Hadamard | $\text{MLP}(h_u \odot h_v)$ | MLP weights |
| Bilinear | $h_u^T W h_v$ | $W \in \mathbb{R}^{d \times d}$ |
| Concatenation | $\text{MLP}([h_u \mathbin\Vert h_v])$ | MLP weights |

The dot product is the most parameter-efficient and a strong default; the experiment in Try it Yourself compares it with a concatenation MLP.

## Negative sampling

You have positive edges (real connections). You need negative examples (non-edges) for binary
cross-entropy training. **Uniform random negative sampling**: for each positive edge $(u,v)$,
sample $K$ random nodes $w$ and treat $(u,w)$ as negative (assuming the graph is sparse enough
that randomly sampled pairs are unlikely to be true edges).

Everything below is plain PyTorch on a synthetic graph (`torch_geometric` is not installed and Cora would need a download): a stochastic block model with 5 communities of 60 nodes (within-community edge probability 0.12, across 0.01) and 100 noisy bag-of-words features.

```python
import numpy as np
import torch
torch.set_num_threads(1)

def make_graph(n_per=60, k=5, p_in=0.06, p_out=0.015, d=100, seed=2):
    """Stochastic block model + noisy bag-of-words features (a tiny Cora stand-in)."""
    rng = np.random.default_rng(seed)
    y = np.repeat(np.arange(k), n_per)
    n = len(y)
    P = np.where(y[:, None] == y[None, :], p_in, p_out)
    U = np.triu(rng.random((n, n)) < P, 1)
    A = U | U.T
    base = np.full((k, d), 0.03)
    for c in range(k):
        base[c, c * 20:(c + 1) * 20] = 0.15        # class-favored words
    X = (rng.random((n, d)) < base[y]).astype(np.float32)
    X = X / np.maximum(X.sum(1, keepdims=True), 1)
    train = np.zeros(n, bool)
    for c in range(k):
        train[np.nonzero(y == c)[0][:10]] = True
    rest = np.nonzero(~train)[0]
    rng.shuffle(rest)
    val = np.zeros(n, bool); val[rest[:100]] = True
    test = ~train & ~val
    return (torch.tensor(X), torch.tensor(np.stack(np.nonzero(A))), torch.tensor(y),
            torch.tensor(train), torch.tensor(val), torch.tensor(test))

x, edge_index, y, train_mask, val_mask, test_mask = make_graph(p_in=0.12, p_out=0.01)
num_nodes, num_classes = x.shape[0], 5

import torch.nn as nn
import torch.nn.functional as F

# Unique undirected edges (u < v)
und = edge_index[:, edge_index[0] < edge_index[1]]
E = und.shape[1]

# Split: 85% train (message passing + supervision), 5% val, 10% test positive edges
g = torch.Generator().manual_seed(0)
perm = torch.randperm(E, generator=g)
n_val, n_test = int(0.05 * E), int(0.10 * E)
val_pos, test_pos = und[:, perm[:n_val]], und[:, perm[n_val:n_val + n_test]]
train_pos = und[:, perm[n_val + n_test:]]
train_ei = torch.cat([train_pos, train_pos.flip(0)], dim=1)   # symmetric message-passing graph

print(f"Train edges (pos): {train_pos.shape[1]}")
print(f"Val  edges (pos):  {val_pos.shape[1]}")
print(f"Test edges (pos):  {test_pos.shape[1]}")
```

## Manual negative sampling

A negative sampler draws random node pairs and rejects self-pairs and real edges. Real edges include the held-out validation and test edges, otherwise a "negative" could secretly be a positive.

```python
A_full = torch.zeros(num_nodes, num_nodes, dtype=torch.bool)
A_full[edge_index[0], edge_index[1]] = True       # every real edge, incl. held-out ones

def my_negative_sampling(num_samples, forbidden, gen):
    """Sample random non-edges (u, v), u != v, not in `forbidden`."""
    out = []
    while sum(o.shape[1] for o in out) < num_samples:
        cand = torch.randint(0, num_nodes, (2, 4 * num_samples), generator=gen)
        ok = (cand[0] != cand[1]) & ~forbidden[cand[0], cand[1]]
        out.append(cand[:, ok])
    return torch.cat(out, dim=1)[:, :num_samples]

gen = torch.Generator().manual_seed(1)
neg = my_negative_sampling(100, A_full, gen)
print(f"Sampled {neg.shape[1]} negative edges")
overlap = int(A_full[neg[0], neg[1]].sum())
print(f"Overlap with real edges: {overlap} (should be 0)")

# Fixed evaluation negatives: one per held-out positive
val_neg = my_negative_sampling(val_pos.shape[1], A_full, gen)
test_neg = my_negative_sampling(test_pos.shape[1], A_full, gen)
```

```output
Sampled 100 negative edges
Overlap with real edges: 0 (should be 0)
```

## Building the link predictor

```python
class GCNLayer(nn.Module):
    def __init__(self, i, o):
        super().__init__(); self.lin = nn.Linear(i, o)
    def forward(self, h, ei):
        n = h.shape[0]; loops = torch.arange(n)
        src, dst = torch.cat([ei[0], loops]), torch.cat([ei[1], loops])
        deg = torch.zeros(n).index_add_(0, dst, torch.ones(len(dst)))
        norm = (deg[src] * deg[dst]).pow(-0.5)
        return torch.zeros(n, self.lin.out_features).index_add_(0, dst, norm[:, None] * self.lin(h)[src])

class LinkPredictor(nn.Module):
    def __init__(self, in_ch, hidden, decoder='dot'):
        super().__init__()
        self.conv1, self.conv2 = GCNLayer(in_ch, hidden), GCNLayer(hidden, hidden)   # GNN encoder
        self.decoder = decoder
        if decoder == 'mlp':
            self.mlp = nn.Sequential(nn.Linear(2 * hidden, hidden), nn.ReLU(), nn.Linear(hidden, 1))

    def encode(self, x, edge_index):
        x = F.relu(self.conv1(x, edge_index))
        x = F.dropout(x, 0.5, self.training)
        return self.conv2(x, edge_index)

    def decode(self, z, edge_index):
        if self.decoder == 'mlp':   # concatenation scorer
            return self.mlp(torch.cat([z[edge_index[0]], z[edge_index[1]]], dim=-1)).squeeze(-1)
        return (z[edge_index[0]] * z[edge_index[1]]).sum(dim=-1)   # dot-product scoring

torch.manual_seed(0)
model = LinkPredictor(x.shape[1], 32)
n_params = sum(p.numel() for p in model.parameters())
print(f"Encoder + dot-product decoder: {n_params} parameters (the decoder adds none)")
```


## Training with negative sampling

Each epoch draws fresh negatives, one per positive edge, and trains with binary cross-entropy.

```python
def train_lp(model, neg_fn=None, epochs=200, lr=0.01, seed=0):
    gen = torch.Generator().manual_seed(seed)
    neg_fn = neg_fn or (lambda pos: my_negative_sampling(pos.shape[1], A_full, gen))
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    for _ in range(epochs):
        model.train(); opt.zero_grad()
        z = model.encode(x, train_ei)
        neg = neg_fn(train_pos)
        ei = torch.cat([train_pos, neg], dim=1)
        labels = torch.cat([torch.ones(train_pos.shape[1]), torch.zeros(neg.shape[1])])
        loss = F.binary_cross_entropy_with_logits(model.decode(z, ei), labels)
        loss.backward(); opt.step()
    return float(loss)

loss = train_lp(model)
print(f"Final training loss: {loss:.4f}")
```

```output
Final training loss: 0.5558
```

## Evaluation: AUC-ROC and Average Precision

```python
from sklearn.metrics import roc_auc_score, average_precision_score

@torch.no_grad()
def evaluate(model, pos, neg):
    model.eval()
    z = model.encode(x, train_ei)                      # message passing uses training edges only
    ei = torch.cat([pos, neg], dim=1)
    labels = torch.cat([torch.ones(pos.shape[1]), torch.zeros(neg.shape[1])]).numpy()
    out = model.decode(z, ei).sigmoid().numpy()
    return roc_auc_score(labels, out), average_precision_score(labels, out)

val_auc,  val_ap  = evaluate(model, val_pos, val_neg)
test_auc, test_ap = evaluate(model, test_pos, test_neg)
print(f"Val   AUC={val_auc:.4f}  AP={val_ap:.4f}")
print(f"Test  AUC={test_auc:.4f}  AP={test_ap:.4f}")
```

```output
Val   AUC=0.7437  AP=0.6830
Test  AUC=0.7657  AP=0.7290
```

AUC around 0.77 is well above chance (0.5) but far from perfect: with 1216 training edges and noisy features, the encoder can only learn "same community tends to link", so it ranks many same-community non-edges above true edges. The validation and test numbers are close, as they should be when no tuning has been done on either.

## Exercise

Implement **corrupted-head/tail negative sampling**: for each positive edge $(u,v)$, generate negatives by
replacing $u$ with a random node (keeping $v$ fixed), and also by replacing $v$ (keeping $u$ fixed). Compare test AUC
against uniform random negatives, using the same number of negatives per positive.

<details>
<summary>Solution</summary>

```python
def corrupt_head_tail(pos, gen):
    """For each positive (u,v), one negative (rand,v) or (u,rand), chosen at random per edge."""
    E_ = pos.shape[1]
    rand_nodes = torch.randint(0, num_nodes, (E_,), generator=gen)
    flip = torch.rand(E_, generator=gen) < 0.5
    head = torch.where(flip, rand_nodes, pos[0])
    tail = torch.where(flip, pos[1], rand_nodes)
    return torch.stack([head, tail])

res = {'uniform': [], 'corrupt': []}
for seed in range(3):
    for name in res:
        torch.manual_seed(seed)
        m = LinkPredictor(x.shape[1], 32)
        gen_ = torch.Generator().manual_seed(seed)
        fn = (lambda pos: corrupt_head_tail(pos, gen_)) if name == 'corrupt' else None
        train_lp(m, neg_fn=fn, seed=seed)
        res[name].append(evaluate(m, test_pos, test_neg)[0])
for name, v in res.items():
    print(f"{name:8s} mean test AUC over 3 seeds: {sum(v)/3:.4f}")
```

```output
uniform  mean test AUC over 3 seeds: 0.7390
corrupt  mean test AUC over 3 seeds: 0.7494
```

Corrupting the head or tail gives 0.7494 versus 0.7390 for uniform negatives, a gain of about 0.01 that is the same size as the seed-to-seed spread, so it is suggestive rather than conclusive. On a graph this small, a random pair is already almost always a plausible easy negative, and corrupted pairs are only slightly harder because they share one endpoint with a true edge. Harder negatives matter more on large graphs where uniform negatives are trivially easy.
