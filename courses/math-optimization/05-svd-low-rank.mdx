```python
    print(f"A v{i+1} = {A @ Vt[i]}   s{i+1} u{i+1} = {s[i] * U[:, i]}")

print(f"ellipse radii by brute force: longest {radii.max():.4f}, shortest {radii.min():.4f}")

step2 = s[:, None] * step1                         # stretch the axes: an axis-aligned ellipse
step3 = U @ step2                                  # rotate the ellipse into place
print("after V^T: radii all 1?", np.allclose(np.linalg.norm(step1, axis=0), 1),
      "| U S V^T x == A x?", np.allclose(step3, A @ circle))
```

## Truncation and the Eckart–Young–Mirsky theorem

P

### Why nothing beats truncation in the spectral norm

P

```python
rng = np.random.default_rng(1)
m, n, k = 40, 30, 5
Q1, _ = np.linalg.qr(rng.standard_normal((m, n)))
Q2, _ = np.linalg.qr(rng.standard_normal((n, n)))
s_true = 0.8 ** np.arange(n)                       # geometrically decaying spectrum
A = Q1 @ np.diag(s_true) @ Q2.T
U, s, Vt = np.linalg.svd(A, full_matrices=False)
A_k = (U[:, :k] * s[:k]) @ Vt[:k]
print(f"truncated SVD, k={k}: spectral err {np.linalg.norm(A - A_k, 2):.6f} = s_(k+1) {s[k]:.6f}")
print(f"                    Frobenius err {np.linalg.norm(A - A_k):.6f} = sqrt(sum tail s^2) "
      f"{np.sqrt((s[k:] ** 2).sum()):.6f}")

best2, bestF = np.inf, np.inf
for _ in range(2000):                               # random rank-k competitors, each optimally fitted
    Q, _ = np.linalg.qr(rng.standard_normal((m, k)))
    B = Q @ (Q.T @ A)                              # best matrix with column space span(Q)
    best2 = min(best2, np.linalg.norm(A - B, 2))
    bestF = min(bestF, np.linalg.norm(A - B))
print(f"best of 2000 random rank-{k} subspaces: spectral {best2:.6f} | Frobenius {bestF:.6f}")

L, R = rng.standard_normal((m, k)), rng.standard_normal((n, k))
for it in range(200):                               # alternating least squares on ||A - L R^T||_F
    R = np.linalg.lstsq(L, A, rcond=None)[0].T
    L = np.linalg.lstsq(R, A.T, rcond=None)[0].T
print(f"ALS local search after 200 sweeps: Frobenius {np.linalg.norm(A - L @ R.T):.6f}")
```

## The pseudoinverse and minimum-norm least squares

P

### Deriving $A^+$ from the SVD

P

```python
rng = np.random.default_rng(3)
A = rng.standard_normal((6, 2)) @ rng.standard_normal((2, 4))     # 6 x 4, rank 2
b = rng.standard_normal(6)
U, s, Vt = np.linalg.svd(A, full_matrices=False)
print("singular values:", s)
tol = max(A.shape) * np.finfo(float).eps * s[0]
s_inv = np.where(s > tol, 1 / s, 0.0)              # invert only the real singular values
A_plus = Vt.T @ np.diag(s_inv) @ U.T
naive = Vt.T @ np.diag(1 / s) @ U.T                # inverting the rounding-noise ones too
print(f"matches np.linalg.pinv: {np.allclose(A_plus, np.linalg.pinv(A))} | "
      f"x+ matches lstsq: {np.allclose(A_plus @ b, np.linalg.lstsq(A, b, rcond=None)[0])}")
checks = [A @ A_plus @ A - A, A_plus @ A @ A_plus - A_plus, (A @ A_plus).T - A @ A_plus, (A_plus @ A).T - A_plus @ A]
print("Moore-Penrose conditions, max violation:", [f"{np.abs(c).max():.0e}" for c in checks])
x_plus = A_plus @ b
x_other = x_plus + Vt[2:].T @ np.array([1.0, -1.0])     # add a null-space vector
print(f"||x+|| {np.linalg.norm(x_plus):.4f}, residual {np.linalg.norm(A @ x_plus - b):.4f} | "
      f"x+ + null: norm {np.linalg.norm(x_other):.4f}, residual {np.linalg.norm(A @ x_other - b):.4f}")
print(f"without the cutoff: ||x|| = {np.linalg.norm(naive @ b):.2e}")
```

## The condition number is a ratio of singular values

P

```python
rng = np.random.default_rng(4)
n = 50
Q1, _ = np.linalg.qr(rng.standard_normal((n, n)))
Q2, _ = np.linalg.qr(rng.standard_normal((n, n)))
s_c = np.logspace(0, -6, n)                        # sigma from 1 down to 1e-6
M = Q1 @ np.diag(s_c) @ Q2.T
print(f"cond = s_max / s_min = {s_c[0] / s_c[-1]:.3e} | np.linalg.cond {np.linalg.cond(M):.3e}")
b = Q1[:, 0]                                        # b along u_1 (the easy direction)
x = np.linalg.solve(M, b)
for name, d in [("worst: delta along u_n", Q1[:, -1]), ("random delta", rng.standard_normal(n))]:
    db = 1e-8 * d / np.linalg.norm(d)
    dx = np.linalg.solve(M, b + db) - x
    amp = (np.linalg.norm(dx) / np.linalg.norm(x)) / (np.linalg.norm(db) / np.linalg.norm(b))
    print(f"{name:24} relative error amplification {amp:.3e}")
```

## Truncated SVD and ridge are both spectral filters

P

### Ridge in SVD coordinates

P

```python
rng = np.random.default_rng(5)
N, p = 100, 30
Q1, _ = np.linalg.qr(rng.standard_normal((N, p)))
Q2, _ = np.linalg.qr(rng.standard_normal((p, p)))
s_x = np.logspace(1, -3, p)                        # cond(X) = 1e4: nearly collinear features
X = Q1 @ np.diag(s_x) @ Q2.T
beta_true = Q2 @ (1.0 / (1 + np.arange(p)))        # most signal on the leading directions
y = X @ beta_true + 0.05 * rng.standard_normal(N)
U, s, Vt = np.linalg.svd(X, full_matrices=False)
coef = U.T @ y / s                                 # OLS coefficient along each v_i

def filtered(f):                                   # beta = sum_i f_i (u_i^T y / s_i) v_i
    return Vt.T @ (f * coef)

print(f"{'method':>16} {'||beta - beta*||':>17} {'train RMSE':>11}")
fits = {"OLS": np.ones(p)}
fits.update({f"TSVD k={k}": (np.arange(p) < k).astype(float) for k in (5, 10, 15, 20)})
fits.update({f"ridge lam={lam:g}": s ** 2 / (s ** 2 + lam) for lam in (1e-4, 1e-2, 1)})
for name, f in fits.items():
    bh = filtered(f)
    print(f"{name:>16} {np.linalg.norm(bh - beta_true):>17.4f} {np.sqrt(np.mean((X @ bh - y) ** 2)):>11.4f}")
lam = 1e-2
ridge_direct = np.linalg.solve(X.T @ X + lam * np.eye(p), X.T @ y)
print("filter form equals (X^T X + lam I)^-1 X^T y:", np.allclose(filtered(s ** 2 / (s ** 2 + lam)), ridge_direct))
print("ridge filter factors at lam=1e-2 for s =", s[[0, 9, 14, 19, 29]].round(4), "->",
      (s ** 2 / (s ** 2 + lam))[[0, 9, 14, 19, 29]].round(4))
```

## Low rank in real data: completing a ratings matrix

P

```python
rng = np.random.default_rng(6)
n_users, n_items, r_true = 300, 120, 4
R_full = 3.5 + rng.standard_normal((n_users, r_true)) @ rng.standard_normal((r_true, n_items)) / 2 \
         + 0.3 * rng.standard_normal((n_users, n_items))
mask = rng.random((n_users, n_items)) < 0.25        # only 25% of ratings observed
test = (~mask) & (rng.random((n_users, n_items)) < 0.2)
rmse = lambda P: np.sqrt(np.mean((P[test] - R_full[test]) ** 2))
item_mean = np.nanmean(np.where(mask, R_full, np.nan), axis=0)
print(f"baseline item-mean RMSE {rmse(np.tile(item_mean, (n_users, 1))):.4f}")

def hard_impute(k, iters=100):
    Z = np.where(mask, R_full, item_mean)          # start: fill missing with item means
    for _ in range(iters):
        U, s, Vt = np.linalg.svd(Z, full_matrices=False)
        L = (U[:, :k] * s[:k]) @ Vt[:k]            # project onto rank k (Eckart-Young)
        Z = np.where(mask, R_full, L)              # keep what you know, impute the rest
    return L

for k in (1, 2, 4, 8, 16):
    print(f"hard-impute rank {k:>2}: test RMSE {rmse(hard_impute(k)):.4f}")
```

## Randomized SVD: sketch, then decompose a small matrix

P

### The Halko–Martinsson–Tropp algorithm

P

```python
from scipy.sparse.linalg import svds
from sklearn.utils.extmath import randomized_svd

def rsvd(A, k, p=10, q=0, seed=0):
    G = np.random.default_rng(seed).standard_normal((A.shape[1], k + p))
    Q, _ = np.linalg.qr(A @ G)                     # sketch the range of A
    for _ in range(q):                             # power iterations sharpen the spectrum
        Q, _ = np.linalg.qr(A.T @ Q)               # re-orthonormalize each pass
        Q, _ = np.linalg.qr(A @ Q)
    Ub, sb, Vtb = np.linalg.svd(Q.T @ A, full_matrices=False)   # (k+p) x n: cheap
    return (Q @ Ub)[:, :k], sb[:k], Vtb[:k]

rng = np.random.default_rng(7)
m, n, k = 2000, 600, 20
Qa, _ = np.linalg.qr(rng.standard_normal((m, n)))
Qb, _ = np.linalg.qr(rng.standard_normal((n, n)))
s_A = 1.0 / np.sqrt(1 + np.arange(n))              # slow decay: the hard case for sketching
A = (Qa * s_A) @ Qb.T
print(f"optimal rank-{k} spectral error s_(k+1) = {s_A[k]:.4f}")
print(f"{'p':>3} {'q':>2} {'||A - U S V^T||_2':>18} {'ratio to optimal':>17} {'products with A':>16}")
for p, q in [(0, 0), (10, 0), (10, 1), (10, 2), (10, 4)]:
    Ur, sr, Vtr = rsvd(A, k, p, q)
    err = linalg.svdvals(A - (Ur * sr) @ Vtr)[0]
    print(f"{p:>3} {q:>2} {err:>18.4f} {err / s_A[k]:>17.3f} {(k + p) * (2 * q + 2):>16,d}")
s_svds = np.sort(svds(A, k=k, random_state=0)[1])[::-1]
s_skl = randomized_svd(A, k, n_oversamples=10, n_iter=4, random_state=0)[1]
print(f"top-{k} singular values, max abs diff vs exact: rsvd(q=4) {np.abs(rsvd(A, k, 10, 4)[1] - s_A[:k]).max():.1e}"
      f" | scipy svds {np.abs(s_svds - s_A[:k]).max():.1e} | sklearn {np.abs(s_skl - s_A[:k]).max():.1e}")
```

```python
d = 4096
print(f"{'rank r':>7} {'LoRA params 2dr':>16} {'full d^2':>12} {'fraction':>9}")
for r in (4, 8, 16, 64):
    print(f"{r:>7} {2 * d * r:>16,d} {d * d:>12,d} {2 * d * r / d**2:>9.4%}")
print(f"7B-style model, 32 layers, LoRA r=8 on W_q and W_v: {32 * 2 * 2 * d * 8:,d} trainable parameters")

rng = np.random.default_rng(8)
dm = 256
W0 = rng.standard_normal((dm, dm)) / np.sqrt(dm)
dW = (rng.standard_normal((dm, 64)) * 0.7 ** np.arange(64)) @ rng.standard_normal((64, dm)) / dm  # task update
Ud, sd, Vtd = np.linalg.svd(dW)
for r in (1, 4, 8, 16):
    B, A_lo = Ud[:, :r] * sd[:r], Vtd[:r]          # best rank-r factors: B (d x r), A (r x d)
    left = np.linalg.norm(dW - B @ A_lo) / np.linalg.norm(dW)
```

```python
def soft_impute(lam, iters=200):
```
